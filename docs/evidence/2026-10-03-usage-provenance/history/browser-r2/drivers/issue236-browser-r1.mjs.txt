assert.equal(process.env.ECORP_USAGE_PROVENANCE_TEST, '1')
const setupPath = process.env.ECORP_USAGE_PROVENANCE_SETUP
assert.ok(path.isAbsolute(setupPath ?? ''))
const fixture = JSON.parse(await readFile(setupPath, 'utf8'))
assert.equal(fixture.test_owned, true)
assert.equal(fixture.provider_fixture, 'native-codex-usage-corpus')
assert.ok(path.isAbsolute(fixture.qa_root) && path.isAbsolute(fixture.source))
assert.match(path.basename(fixture.qa_root), /^issue236-usage-provenance-[a-z0-9-]+$/iu)
assert.equal(path.resolve(await realpath(fixture.qa_root)), path.resolve(fixture.qa_root))
assert.equal(path.basename(path.dirname(fixture.qa_root)).toLowerCase(), 'qa')
const output = path.join(fixture.qa_root, 'evidence')
const corpusBytes = await readFile(new URL('./issue236-acceptance-corpus-r1.json', import.meta.url))
const corpusHash = createHash('sha256').update(corpusBytes).digest('hex')
assert.equal(corpusHash, fixture.corpus_sha256)

function owned(value) {
  assert.ok(path.isAbsolute(value))
  const relative = path.relative(fixture.qa_root, value)
  assert.ok(relative && relative !== '..' && !relative.startsWith('..' + path.sep) && !path.isAbsolute(relative))
  return value
}
owned(fixture.source)
assert.notEqual(path.resolve(fixture.source), path.resolve(fixture.repository))
function origin(value) {
  const url = new URL(value)
  assert.equal(url.protocol, 'http:')
  assert.equal(url.hostname, '127.0.0.1')
  assert.ok(url.port && !url.username && !url.password && !url.search && !url.hash && url.pathname === '/')
  return url.origin
}
const server = origin(fixture.server_url), web = origin(fixture.web_url)
const allowed = new Set([server, web])
const allowedWs = new Set([server.replace('http:', 'ws:'), web.replace('http:', 'ws:')])
const demo = fixture.demo
const api = suffix => '/api/corps/' + demo.corp_id + suffix
const report = { schema_version: 1, issue: 236, status: 'running', started_at_utc: new Date().toISOString(),
  source_binding: fixture.source_binding, corpus_sha256: corpusHash, assertions: [], scenarios: [],
  http_results: [], blocked_requests: [], page_errors: [], screenshots: [],
  assurance: 'Owned deterministic transport in actual Edge, native server/runner and fresh SCRAM PostgreSQL. No provider inference, billing verification, auditor decision, hosted gate or independent outcome review.' }
const save = () => writeFile(reportPath, JSON.stringify(report, null, 2) + '\n')
function verify(name, action) {
  try { action(); report.assertions.push({ name, passed: true }) }
  catch (error) { report.assertions.push({ name, passed: false, error: error.message }); throw error }
}
async function request(route, body, method = body === undefined ? 'GET' : 'POST') {
  assert.ok(route.startsWith('/api/'))
  const response = await fetch(server + route, { method, redirect: 'error',
    headers: body === undefined ? undefined : { 'content-type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body), signal: AbortSignal.timeout(10_000) })
  if (method !== 'GET' || response.status !== 200) report.http_results.push({ route, method, status: response.status })
  assert.equal(response.status, 200, route + ': unexpected HTTP status')
  return response.json()
}
const snapshot = () => request(api('/snapshot?actor_id=' + demo.alice_actor_id))
function git(args) {
  return execFileSync('git', ['--no-optional-locks', '-C', fixture.source, ...args],
    { encoding: 'utf8', windowsHide: true, timeout: 10_000 }).trim()
}
function sourcePreserved() {
  assert.equal(git(['rev-parse', 'HEAD']), fixture.source_commit)
  assert.equal(git(['status', '--porcelain=v1', '--untracked-files=all', '--ignored']), '')
}
const deadline = Date.now() + 10 * 60_000
async function waitFor(read, predicate, timeout = 90_000) {
  const until = Math.min(deadline, Date.now() + timeout)
  while (Date.now() < until) {
    const value = await read()
    if (predicate(value)) return value
    await delay(125)
  }
  throw new Error('Owned usage acceptance reached its observation ceiling')
}
let browser, page, active
const terminal = run => run && ['completed', 'failed', 'cancelled', 'lost'].includes(run.status)
async function screenshot(name) {
  const destination = path.join(output, name)
  await page.screenshot({ path: destination, fullPage: true })
  report.screenshots.push({ path: destination,
    sha256: createHash('sha256').update(await readFile(destination)).digest('hex') })
}
try {
  await save()
  verify('Configured synthetic source starts clean at the recorded commit', sourcePreserved)
  const require = createRequire(import.meta.url)
  const { chromium } = require(process.env.CRONY_PLAYWRIGHT_MODULE)
  browser = await chromium.launch({ channel: 'msedge', headless: true })
  const context = await browser.newContext({ viewport: { width: 1440, height: 1050 }, reducedMotion: 'reduce' })
  await context.route('**/*', async route => {
    const url = new URL(route.request().url())
    if (allowed.has(url.origin)) await route.continue()
    else {
      report.blocked_requests.push({ protocol: url.protocol, hostname: url.hostname })
      await route.abort('blockedbyclient')
    }
  })
  await context.routeWebSocket('**/*', socket => {
    const url = new URL(socket.url())
    if (allowedWs.has(url.origin)) socket.connectToServer()
    else {
      report.blocked_requests.push({ protocol: url.protocol, hostname: url.hostname })
      socket.close()
    }
  })
  page = await context.newPage()
  page.setDefaultTimeout(20_000)
  page.on('pageerror', () => report.page_errors.push('Browser pageerror'))
  for (const scenario of corpus.cases) {
    assert.ok(Date.now() < deadline)
    const current = { id: scenario.id, status: 'running', expected: scenario.expected }
    report.scenarios.push(current)
    active = current
    const marker = '[issue236:' + scenario.id + ']'
    current.title = 'Usage provenance ' + scenario.id
    const created = await request(api('/missions'), {
      requested_by: demo.alice_actor_id, preferred_adapter: 'codex', strategy: 'single',
      title: current.title, description: marker + ' Replay only the frozen synthetic usage corpus. Do not change source or call external services.',
      source: { repository: fixture.source_repository, base_ref: 'main', base_commit: fixture.source_commit },
      budget_tokens: 100_000, budget_cost_microusd: 10_000_000,
    })
    current.mission_id = created.mission_id
    await page.goto(web + '/#missions', { waitUntil: 'networkidle' })
    await page.locator('.live-indicator.live-live').waitFor()
    await page.getByRole('button').filter({ hasText: current.title }).click()
    const card = page.locator('[data-mission-id="' + current.mission_id + '"]')
    const launchResponse = page.waitForResponse(response => response.url() ===
      server + api('/missions/' + current.mission_id + '/launch') && response.request().method() === 'POST')
    await card.getByTestId('work-result-card').getByRole('button', { name: 'Start mission', exact: true }).click()
    const launch = await launchResponse
    verify(scenario.id + ': browser launches through the authoritative server', () => assert.equal(launch.status(), 200))
    current.run_id = (await launch.json()).run_id
    assert.ok(current.run_id)
    await save()
    const settled = await waitFor(snapshot, state => {
      const run = state.snapshot.runs.find(item => item.id === current.run_id)
      return terminal(run) && ['removed', 'preserved'].includes(run.workspace_disposition)
    })
    const run = settled.snapshot.runs.find(item => item.id === current.run_id)
    const events = settled.snapshot.events.filter(item => item.aggregate_id === current.run_id)
    const task = settled.snapshot.tasks.find(item => item.id === run.task_id)
    current.agent_id = run.agent_id
    current.run = { id: run.id, task_id: run.task_id, status: run.status, workspace_path: run.workspace_path,
      workspace_disposition: run.workspace_disposition, input_tokens: run.input_tokens,
      output_tokens: run.output_tokens, cost_microusd: run.cost_microusd, breaker_stage: run.breaker_stage,
      provider_session_id: run.provider_session_id }
    current.usage_events = events.filter(item => item.type === 'run.usage' || item.type === 'run.usage_observed')
      .map(item => ({ id: item.id, type: item.type, aggregate_id: item.aggregate_id, payload: item.payload }))
    current.completion_events = events.filter(item => ['run.verification_passed', 'run.completed',
      'run.workspace_removed', 'run.workspace_preserved', 'run.failed', 'run.verification_failed'].includes(item.type))
      .map(item => ({ id: item.id, type: item.type, aggregate_id: item.aggregate_id, payload: item.payload }))
    await save()
    verify(scenario.id + ': persisted verifier precedes accepted task and run completion', () => {
      assert.equal(run.status, 'completed')
      assert.equal(task.status, 'completed')
      assert.equal(settled.snapshot.runs.filter(item => item.task_id === task.id).length, 1)
      assert.equal(events.filter(item => item.type === 'run.completed').length, 1)
      assert.ok(events.some(item => item.type === 'run.verification_passed'))
      assert.ok(!events.some(item => ['run.failed', 'run.verification_failed'].includes(item.type)))
    })
    verify(scenario.id + ': native counts and unknown quantities survive the full journal path', () => {
      const usage = current.usage_events
      assert.equal(usage.length, scenario.expected.emitted)
      for (const event of usage) {
        assert.equal(event.type, 'run.usage')
        assert.equal(event.payload.usage_validation.policy, 'native_usage_v1')
        assert.equal(event.payload.usage_validation.disposition, 'accepted')
        assert.equal(event.payload.accounting.charged, true)
        assert.equal(event.payload.accounting.known_subtotal_only, true)
        assert.equal(event.payload.cost_microusd, null)
        assert.equal(event.payload.usage_coverage.tokens, scenario.expected.token_coverage)
        assert.equal(event.payload.usage_coverage.usd, 'unavailable')
        assert.equal(event.payload.usage_coverage.complete_provider_bill, false)
        assert.equal(event.payload.usage_provenance.mapping, 'codex_app_server_last_v1')
        assert.equal(event.payload.usage_provenance.provider_session_id, run.provider_session_id)
        assert.equal(event.payload.usage_provenance.turn_id, 'turn-1')
        assert.ok(event.payload.usage_identity_keys.length > 0)
      }
      assert.equal(run.input_tokens, scenario.expected.input_tokens ?? 0)
      assert.equal(run.output_tokens, scenario.expected.output_tokens ?? 0)
      assert.equal(run.cost_microusd, 0)
      if (scenario.expected.input_tokens === null) {
        assert.equal(usage[0].payload.input_tokens, null)
        assert.equal(usage[0].payload.output_tokens, 0)
      }
      if (scenario.id === 'codex-cache-and-reasoning-subsets') {
        const provenance = usage[0].payload.usage_provenance
        assert.equal(provenance.cached_input_tokens, 40)
        assert.equal(provenance.cache_write_input_tokens, 10)
        assert.equal(provenance.reasoning_output_tokens, 5)
        assert.equal(run.input_tokens + run.output_tokens, 120)
      }
    })
    verify(scenario.id + ': assigned clean worktree is distinct from the preserved configured source', () => {
      owned(run.workspace_path)
      assert.notEqual(path.resolve(run.workspace_path), path.resolve(fixture.source))
      assert.equal(run.workspace_disposition, 'removed')
      sourcePreserved()
    })
    await assert.rejects(lstat(run.workspace_path), { code: 'ENOENT' })
    await page.goto(web + '/#missions', { waitUntil: 'networkidle' })
    await page.locator('.live-indicator.live-live').waitFor()
    await page.getByRole('button').filter({ hasText: current.title }).click()
    const settledCard = page.locator('[data-mission-id="' + current.mission_id + '"]')
    await settledCard.waitFor()
    current.visible_result = await settledCard.getByTestId('work-result-card').innerText()
    await screenshot('usage-' + scenario.id + '.png')
    current.status = 'passed'
    await save()
    active = null
  }
  verify('Owned browser made no out-of-scope requests and raised no page errors', () => {
    assert.deepEqual(report.blocked_requests, [])
    assert.deepEqual(report.page_errors, [])
  })
  report.status = 'passed'
} catch (error) {
  report.status = 'failed'
  report.failure = error.message
  if (active) active.status = 'failed'
  if (page) await screenshot('usage-failure.png').catch(() => {})
  if (active?.run_id) {
    try {
      const state = await snapshot()
      const run = state.snapshot.runs.find(item => item.id === active.run_id)
      if (run && !terminal(run)) {
        await request(api('/agents/' + run.agent_id + '/emergency-stop'), {
          actor_id: demo.alice_actor_id, reason: 'Owned usage fixture failed; stop this exact fixture run.',
        })
      }
    } catch (cleanupError) { report.failure_stop_error = cleanupError.message }
  }
} finally {
  if (browser) await browser.close().catch(error => { report.status = 'failed'; report.browser_close_error = error.message })
  report.finished_at_utc = new Date().toISOString()
  report.counts = { passed: report.assertions.filter(item => item.passed).length,
    failed: report.assertions.filter(item => !item.passed).length,
    scenarios_passed: report.scenarios.filter(item => item.status === 'passed').length,
    scenarios_unexecuted: corpus.cases.length - report.scenarios.length }
  await save()
  console.log(JSON.stringify({ status: report.status, counts: report.counts, failure: report.failure }))
  process.exitCode = report.status === 'passed' ? 0 : 1
}
