async function sql(label, statement, mutation = false) {
  assert.ok(!frozen || !mutation, 'No further SQL mutation after the eligibility boundary')
  const env = Object.fromEntries(Object.entries(process.env).filter(([name]) => !/^PG|^(DATABASE_URL|GH_|GITHUB_|AZURE_|CRONY_|ECORP_)/i.test(name)))
  env.PGPASSFILE = path.join(setup.qa_root, 'credentials', 'pgpass.conf')
  const started = new Date().toISOString()
  const result = await new Promise((resolve, reject) => {
    const child = spawn(setup.psql, ['-X', '-w', '-q', '-t', '-A', '-v', 'ON_ERROR_STOP=1', '-h', setup.database.host,
      '-p', String(setup.database.port), '-U', setup.database.user, '-d', setup.database.name], { env, windowsHide: true, stdio: ['pipe', 'pipe', 'pipe'] })
    let stdout = '', stderr = ''
    const timer = setTimeout(() => { child.kill(); reject(new Error('Owned SQL operation timed out: ' + label)) }, 15000)
    child.stdout.setEncoding('utf8'); child.stderr.setEncoding('utf8')
    child.stdout.on('data', bytes => { stdout += bytes })
    child.stderr.on('data', bytes => { stderr += bytes })
    child.on('error', error => { clearTimeout(timer); reject(error) })
    child.on('close', code => { clearTimeout(timer); resolve({ code, stdout, stderr }) })
    child.stdin.end(statement + '\n')
  })
  const entry = { label, mutation, started_at: started, finished_at: new Date().toISOString(), exit_code: result.code,
    statement_sha256: sha256(statement), statement, stderr: result.stderr }
  report.sql_operations.push(entry)
  await save()
  assert.equal(result.code, 0, label + ': ' + result.stderr.slice(0, 1500))
  const values = result.stdout.trim().split(/\r?\n/).filter(Boolean).map(line => JSON.parse(line))
  assert.equal(values.length, 1, label + ' must return one JSON value')
  entry.result = values[0]
  return values[0]
}

async function until(label, read, predicate, timeout = 120000) {
  const deadline = Date.now() + timeout
  while (Date.now() < deadline) {
    const value = await read()
    if (predicate(value)) return value
    await delay(250)
  }
  throw new Error('Timed out waiting for ' + label)
}
function view(state, missionId) {

}

async function complete(record) {
  const source = setup.sources[record.side]
  const final = await until(record.side + ' verified native completion', async () => view(await snapshot(), record.mission_id), value =>
    value.runs.length === 1 && ['completed', 'failed'].includes(value.mission.status) && value.runs[0].workspace_disposition === 'preserved')
  const run = final.runs[0]
  record.final = { mission: final.mission, tasks: final.tasks, run,
    evidence: final.state.verification_evidence.filter(item => item.run_id === run.id).sort((a, b) => a.check_index - b.check_index),
    events: final.state.events.filter(item => item.aggregate_id === run.id) }
  await save()
  assert.equal(final.mission.status, 'completed'); assert.equal(run.status, 'completed')
  assert.equal(run.runner_id, source.runner_id); assert.equal(run.source_base_commit, source.base_commit)
  assert.deepEqual(final.tasks[0].verification_policy, policy)
  assert.equal(record.final.evidence.length, 2)
  assert.ok(record.final.evidence.every(item => item.status === 'passed'))
  assert.equal(record.final.evidence[1].payload.exit_code, 0)
  assert.deepEqual(record.final.evidence[1].payload.args, ['verify.mjs'])
  assert.equal(final.state.verification_requests.filter(row => row.run_id === run.id).length, 0)
  const relative = path.relative(source.workspace, run.workspace_path)
  assert.ok(relative && !relative.startsWith('..') && !path.isAbsolute(relative))
  const outputBytes = await readFile(path.join(run.workspace_path, 'output.txt'))
  assert.deepEqual(outputBytes, Buffer.from(`Issue 172 native runner ${record.side} completed.\n`))
  const workspaceStatus = (await git(run.workspace_path, ['status', '--porcelain=v1', '--untracked-files=all'])).toString()
  assert.equal(workspaceStatus, '?? result.md\n')
  const diagnostic = await readFile(path.join(run.workspace_path, 'result.md'))
  assert.deepEqual(diagnostic, Buffer.from(`# Native issue172 diagnostic\n\nRunner fixture: ${record.side}\nRun: ${run.id}\nThe persisted verifier decides completion.\n`))
  const preserved = record.final.events.find(item => item.type === 'run.workspace_preserved')
  assert.ok(preserved)
  assert.equal(preserved.payload.dirty, true)
  assert.equal(preserved.payload.workspace, run.workspace_path)
  record.preserved_diagnostic = { path: path.join(run.workspace_path, 'result.md'), sha256: sha256(diagnostic), workspace_status: workspaceStatus,
    scope: 'Exact deterministic provider diagnostic remains untracked and is preserved; no other dirty files are allowed.' }

  assert.equal(sha256(bytes), delivered.sha256)
  assert.equal(envelope.head_commit, delivered.head_commit)
  assert.equal(envelope.base_commit, source.base_commit)
  assert.equal(envelope.verification_sha256, run.verification_sha256)
  assert.equal(run.deliverable_sha256, delivered.sha256)
  assert.equal(envelope.source_verification.tree, envelope.verified_tree)

    WHERE corp_id=${uuid(demo.corp_id)} AND seq>${Number(report.boundary.event_seq)} ORDER BY seq) e;`)
  const first = report.events_after_boundary[0]
  assert.equal(first.type, 'run.requested'); assert.equal(first.aggregate_id, bFinal.runs[0].id)
  assert.equal(first.payload.runner_id, 'b-issue172-native-qa'); assert.equal(first.payload.mission_launch, false)
  report.dispatch_latency_ms = Date.parse(first.created_at) - Date.parse(report.boundary.after.updated_at)
  assert.ok(report.dispatch_latency_ms >= 0 && report.dispatch_latency_ms < 15000)
  assert.equal(report.events_after_boundary.filter(item => /^runner\.(connected|disconnected|registered|revoked)/.test(item.type)).length, 0)
  report.assignment_fences = await sql('non-secret assignment fence assertions', `SELECT jsonb_build_object(
    'A_has_assignment',(SELECT assignment_token IS NOT NULL FROM runs WHERE id=${uuid(aFinal.runs[0].id)}),
    'B_has_assignment',(SELECT assignment_token IS NOT NULL FROM runs WHERE id=${uuid(bFinal.runs[0].id)}),
    'assignments_distinct',(SELECT a.assignment_token<>b.assignment_token FROM runs a,runs b WHERE a.id=${uuid(aFinal.runs[0].id)} AND b.id=${uuid(bFinal.runs[0].id)}),
    'B_same_corp',(SELECT corp_id=${uuid(demo.corp_id)} FROM runs WHERE id=${uuid(bFinal.runs[0].id)}),
    'B_expected_runner',(SELECT runner_id='b-issue172-native-qa' FROM runs WHERE id=${uuid(bFinal.runs[0].id)}));`)
  assert.ok(Object.values(report.assignment_fences).every(Boolean))
  assert.equal(report.browser_mutations.length, report.browser_mutations_at_boundary)
  for (const source of Object.values(setup.sources)) {
    assert.equal((await git(source.path, ['rev-parse', 'HEAD'])).toString().trim(), source.base_commit)
    assert.equal((await git(source.path, ['status', '--porcelain=v1', '--untracked-files=all'])).toString(), '')
  }
  assert.equal(report.errors.length, 0); assert.equal(report.blocked_requests.length, 0)
  report.source_unchanged = true; report.status = 'passed'
} catch (error) {
  report.status = 'failed'
  report.failure = String(error.message).replace(/\bpostgres(?:ql)?:\/\/\S+/g, '[database URL withheld]').slice(0, 2500)
  if (page) report.failure_screenshot = await screenshot('failure').catch(() => null)
  process.exitCode = 1
} finally {
  await browser?.close()
  report.finished_at = new Date().toISOString()
  await save()
  console.log(JSON.stringify({ status: report.status, stage: report.stage, report: reportPath, failure: report.failure }))
}
