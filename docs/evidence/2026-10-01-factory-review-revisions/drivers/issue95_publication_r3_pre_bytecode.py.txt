def now():
    return datetime.now(timezone.utc).isoformat()


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def compact(value):
    return json.dumps(value, separators=(",", ":"), ensure_ascii=False).encode()


def load(path):
    return json.loads(Path(path).read_text(encoding="utf-8-sig"))


def write(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(value, indent=2, ensure_ascii=False) + "\n")


def git(*args, env=ENV, data=None, timeout=240):
    result = subprocess.run(["git", "--no-optional-locks", "-C", str(ROOT), *args],
                            env=env, input=data, capture_output=True, timeout=timeout,
                            creationflags=subprocess.CREATE_NO_WINDOW)
    if result.returncode:
        raise RuntimeError(f"Git {args[0]} exited {result.returncode}")
    return result.stdout


def index_digest():
    path = Path(git("rev-parse", "--path-format=absolute", "--git-path", "index").decode().strip())
    return sha(path.read_bytes())

def guarded_bytes(path, root):
    declared_root = Path(root).absolute()
    resolved_root = reader.directory_root(declared_root)
    assert os.path.normcase(str(declared_root)) == os.path.normcase(str(resolved_root))
    declared = Path(path).absolute()
    assert declared.is_relative_to(declared_root)
    relative = declared.relative_to(declared_root).as_posix()
    resolved = reader.regular_file(resolved_root, relative)
    assert os.path.normcase(str(declared)) == os.path.normcase(str(resolved))
    before = declared.lstat()
    fields = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_nlink)
    flags = os.O_RDONLY | getattr(os, "O_BINARY", 0) | getattr(os, "O_NOFOLLOW", 0)
    descriptor = os.open(declared, flags)
    with os.fdopen(descriptor, "rb") as stream:
        opened = os.fstat(stream.fileno())
        assert stat.S_ISREG(opened.st_mode) and opened.st_nlink == 1
        assert fields(before) == fields(opened), "source changed before read"
        raw = stream.read()
        assert len(raw) == opened.st_size
        assert fields(os.fstat(stream.fileno())) == fields(opened), "source changed during read"
        assert reader.regular_file(resolved_root, relative) == resolved
        assert fields(declared.lstat()) == fields(opened), "source path replaced during read"
    return raw


def guarded_cargo_binary(path):
    """Read only the three fixed Cargo output/dep pairs, without relaxing evidence guards."""
    root = reader.directory_root(ROOT)
    assert os.path.normcase(str(root)) == os.path.normcase(str(ROOT.absolute()))
    peers = {f"target/debug/crony-{name}.exe": f"target/debug/deps/crony_{name}.exe"
             for name in ("server", "runner", "cli")}
    relative = Path(path).absolute().relative_to(ROOT.absolute()).as_posix()
    assert relative in peers, "Unexpected native build output"
    declared = [root / relative, root / peers[relative]]
    paths = [reader.unlinked_path(item) for item in declared]
    assert all(os.path.normcase(str(a)) == os.path.normcase(str(b)) for a, b in zip(declared, paths))
    fields = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_nlink)
    before = [item.lstat() for item in paths]
    assert all(stat.S_ISREG(info.st_mode) and info.st_nlink == 2 for info in before)
    assert fields(before[0]) == fields(before[1]), "Cargo output/dep identities differ"
    flags = os.O_RDONLY | getattr(os, "O_BINARY", 0) | getattr(os, "O_NOFOLLOW", 0)
    with os.fdopen(os.open(paths[0], flags), "rb") as stream:
        opened = os.fstat(stream.fileno())
        assert stat.S_ISREG(opened.st_mode) and fields(opened) == fields(before[0])
        raw = stream.read()
        assert len(raw) == opened.st_size and fields(os.fstat(stream.fileno())) == fields(opened)
        for declared_path, canonical_path in zip(declared, paths):
            assert reader.unlinked_path(declared_path) == canonical_path
            assert fields(declared_path.lstat()) == fields(opened), "Cargo binary changed during read"
    return raw

@contextmanager
def temporary_index(initial):
    descriptor, name = tempfile.mkstemp(prefix="issue95-publication-index-", dir=EV)
    os.close(descriptor)
    path = Path(name).resolve()
    assert path.parent == EV and path.name.startswith("issue95-publication-index-")
    path.unlink()
    try:
        env = dict(ENV, GIT_INDEX_FILE=str(path), GIT_LITERAL_PATHSPECS="1")
        git("read-tree", initial, env=env)
        yield env
    finally:
        if path.exists():
            assert path.resolve().parent == EV and path.name.startswith("issue95-publication-index-")
            path.unlink()

def sanitize_text(value):
    roots = [(str(ROOT), "<ISOLATED_CHECKOUT>"), (str(EV), "<RUN_EVIDENCE>"),
             (r"<CONFIGURED_SOURCE>", "<CONFIGURED_SOURCE>"),
             (r"<OWNED_QA_PARENT>", "<OWNED_QA_PARENT>")]
    for source, replacement in roots:
        value = value.replace(source, replacement).replace(source.replace("\\", "/"), replacement)
    value = re.sub(r"(?:[a-z]:)?[/\\]+Users[/\\]+[^/\\\s\"'<>]+", "<USER_HOME>", value, flags=re.I)
    value = re.sub(r"\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b",
                   lambda m: "fixture-id-" + sha(m[0].lower().encode())[:16], value, flags=re.I)
    value = re.sub(r"(?i)C\^?:\^?<USER_HOME>\\shyamsridhar", "<USER_HOME>", value)
    value = re.sub(r"\x1b\[[0-?]*[ -/]*[@-~]", "", value).replace("\r\n", "\n").replace("\r", "\n")
    return "\n".join(line.rstrip() for line in value.split("\n"))


def sanitize(value):
    script = ("import {readFileSync} from 'node:fs';import {redactEvidenceCapabilities} from './tools/redact_evidence_capabilities.mjs';"
              "process.stdout.write(JSON.stringify(redactEvidenceCapabilities(JSON.parse(readFileSync(0,'utf8')))));")
    result = subprocess.run([str(NODE), "--input-type=module", "-e", script], cwd=ROOT, env=ENV,
                            input=compact(value), capture_output=True, timeout=120, creationflags=subprocess.CREATE_NO_WINDOW)
    assert result.returncode == 0
    def walk(item):
        if isinstance(item, str):
            return sanitize_text(item)
        if isinstance(item, list):
            return [walk(v) for v in item]
        if isinstance(item, dict):
            return {k: walk(v) for k, v in item.items()}
        return item
    return walk(json.loads(result.stdout))


def projected_receipt(value, physical):
    result = dict(value)
    if isinstance(result.get("source_files"), list):
        manifest = pairs(result.pop("source_files"))
        current = dict(physical)
        result["source_manifest_projection"] = {"recorded_files": len(manifest), "sha256": sha(compact(manifest)),
            "packet_root_manifest": "source-binding.json#/physical_files", "complete_match": manifest == physical,
            "differing_recorded_paths": [n for n, h in manifest if current.get(n) != h],
            "current_paths_not_recorded": sorted(set(current) - set(dict(manifest)))}
    elif "source_files" in result:
        assert type(result["source_files"]) is int and result["source_files"] >= 0, "Unexpected source file count"
    if isinstance(result.get("source"), dict):
        source = dict(result["source"])
        if isinstance(source.get("files"), list) and set(source["files"]) == set(dict(physical)):
            source["files"] = {"packet_root_manifest": "source-binding.json#/physical_files",
                               "projection": "paths only", "recorded_files": len(physical),
                               "original_array_sha256": sha(compact(source["files"]))}
        result["source"] = source
    return result

        stage_names(dict(physical), temporary)
        return git("write-tree", env=temporary).decode().strip()


def bind_blobs(tree, physical):
    mapping = dict(physical)
    proof = []
    for entry in git("ls-tree", "-r", "-z", tree).split(b"\0")[:-1]:
        meta, filename = entry.split(b"\t", 1)
        mode, kind, oid = meta.decode().split()
        name = filename.decode()
        assert kind == "blob" and mode in {"100644", "100755"} and name in mapping, name
        raw = guarded_bytes(ROOT / name, ROOT)
        assert sha(raw) == mapping[name], name
        object_id = lambda data: hashlib.sha1(b"blob " + str(len(data)).encode() + b"\0" + data).hexdigest()
        transform = "identical bytes"
        if object_id(raw) != oid:
            assert object_id(raw.replace(b"\r\n", b"\n")) == oid, "Unexpected Git filter: " + name
            transform = "CRLF to LF only"
        proof.append([name, mode, oid, transform])
    assert len(proof) == len(physical) and {row[0] for row in proof} == set(mapping)
    return sorted(proof)


def plan_for(names):
    script = ("import {readFileSync} from 'node:fs';import {checkPlan,selectNodeTests} from './tools/run_checks.mjs';"
              "const files=JSON.parse(readFileSync(0,'utf8'));"
              "process.stdout.write(JSON.stringify({plan:checkPlan('full',files),node:selectNodeTests(files)}));")
    result = subprocess.run([str(NODE), "--input-type=module", "-e", script], cwd=ROOT, env=ENV,
                            input=compact(sorted(names)), capture_output=True, timeout=180,
                            creationflags=subprocess.CREATE_NO_WINDOW)
    assert result.returncode == 0
    return json.loads(result.stdout)

def make_public_copies(copies, physical, destination):
    """Discover authority-bearing strings in memory, then redact every public text."""
    sensitive = set()
    key_pattern = re.compile(r"capabilit(?:y|ies)|(?:^|_)(?:token|secret|password|credential)(?:$|_)|"
                             r"^(?:authorization|cookie|private_key|signing_key|api_key)$", re.I)

    def collect_strings(value):
        if isinstance(value, str) and len(value) >= 12:
            sensitive.add(value)
        elif isinstance(value, list):
            for item in value:
                collect_strings(item)
        elif isinstance(value, dict):
            for item in value.values():
                collect_strings(item)

    def discover(value):
        if isinstance(value, dict):
            for key, child in value.items():
                if key_pattern.search(key) and not isinstance(child, (dict, list)):
                    collect_strings(child)
                else:
                    discover(child)
        elif isinstance(value, list):
            for child in value:
                discover(child)

    for path, _ in copies:
        if path.suffix == ".json":
            discover(load(path))
    for revision in (6, 7):
        qa = Path(load(EV / f"issue95-acceptance-r{revision}/acceptance.json")["qa_root"])
        assert qa.resolve().is_relative_to(Path(r"<OWNED_QA_PARENT>").resolve())
        for path in [*qa.glob("*.json"), *(qa / "evidence").glob("*.json")]:
            discover(load(path))

    def clean_text(value):
        for secret in sorted(sensitive, key=len, reverse=True):
            value = value.replace(secret, "<REDACTED_AUTHORITY_STRING>")
        return sanitize_text(value)

    def clean(value):
        if isinstance(value, str):
            return clean_text(value)
        if isinstance(value, list):
            return [clean(child) for child in value]
        if isinstance(value, dict):
            return {key: clean(child) for key, child in value.items()}
        return value

    artifacts = []
    try:
        for path, name in copies:
            declared_root = ROOT if path in REPORTS else EV
            raw = guarded_bytes(path, declared_root)
            target = destination / name
            assert target.is_relative_to(destination)
            target.parent.mkdir(parents=True, exist_ok=True)
            if path.suffix == ".png":
                public = raw
                treatment = "Original PNG capture bytes; visually inspected in the implementation pass."
            elif path.suffix == ".json":
                original = json.loads(raw.decode("utf-8-sig"))
                projected = projected_receipt(original, physical)
                public = (json.dumps(clean(sanitize(projected)), indent=2, ensure_ascii=False) + "\n").encode()
                treatment = "Structured capability hashes, stable synthetic UUID aliases, personal-path and authority-string redaction, manifest projection, LF and trailing-whitespace normalization."
            else:
                public = clean_text(raw.decode("utf-8-sig")).encode()
                treatment = "Personal-path and discovered authority-string redaction, stable synthetic UUID aliases, LF and trailing-whitespace normalization."
            with target.open("xb") as stream:
                stream.write(public)
            artifacts.append({"source": path.relative_to(declared_root).as_posix(),
                              "source_root": "reviewed_checkout" if declared_root == ROOT else "run_evidence",
                              "file": name, "original_sha256": sha(raw), "published_sha256": sha(public),
                              "original_bytes": len(raw), "published_bytes": len(public), "treatment": treatment})
    finally:
        sensitive.clear()
    return artifacts

def packet_files():
    return [{"path": path.relative_to(ROOT).as_posix(), "sha256": sha(guarded_bytes(path, ROOT)),
             "bytes": path.stat().st_size}
            for path in sorted(PACKET.rglob("*")) if path.is_file()]


def observed_exit(name):
    receipt = load(EV / name)
    assert receipt["exit_code"] == 0
    assert receipt["observation"] in {"write_stdin returned process exit", "exec_command returned process exit"}
    if receipt["observation"] == "write_stdin returned process exit":
        assert isinstance(receipt["session_id"], int)
    else:
        assert receipt.get("session_id") is None and receipt["tool_return"]["exit_code"] == 0
    # Recording time is deliberately not relabeled as the exact tool-return time.
    timestamp = receipt.get("observed_at_utc") or receipt.get("recorded_at_utc")
    assert timestamp
    assert datetime.fromisoformat(timestamp).utcoffset() is not None
    if "tool_return" in receipt:
        assert receipt["tool_return"]["exit_code"] == receipt["exit_code"]
    if "receipt_sha256" in receipt:
        target = EV / receipt["receipt"]
        assert target.is_relative_to(EV)
        assert sha(guarded_bytes(target, EV)) == receipt["receipt_sha256"]
    return receipt

    return {name: sha(guarded_bytes(ROOT / name, ROOT)) if (ROOT / name).is_file() else None

def scan_secrets(lane, args, record, save, prefix, policy):
    assert sha(guarded_bytes(GITLEAKS, GITLEAKS.parent)) == GITLEAKS_SHA
    version = subprocess.check_output([str(GITLEAKS), "version"], env=ENV).decode().strip()
    assert version == "8.30.1"
    log, report = EV / f"{prefix}-{lane}.log", EV / f"{prefix}-{lane}-redacted.json"
    assert not log.exists() and not report.exists()
    flags = ["--redact=100", "--no-banner", "--no-color", "--ignore-gitleaks-allow",
             "--gitleaks-ignore-path", str(ROOT / ".gitleaksignore"), "--exit-code=42",
             "--timeout=300", "--report-format=json", "--report-path", str(report)]
    if policy[".gitleaks.toml"] is not None:
        flags += ["--config", str(ROOT / ".gitleaks.toml")]
    started = now()
    result = subprocess.run([str(GITLEAKS), *args, *flags], cwd=ROOT, env=ENV,
                            capture_output=True, timeout=360, creationflags=subprocess.CREATE_NO_WINDOW)
    with log.open("xb") as stream:
        stream.write(result.stdout + result.stderr)
    findings = load(report) if report.exists() else None
    item = {"lane": lane, "version": version, "binary_sha256": GITLEAKS_SHA,
            "arguments": [str(GITLEAKS), *args, *flags], "started_at_utc": started,
            "finished_at_utc": now(), "exit_code": result.returncode,
            "finding_count": len(findings) if isinstance(findings, list) else None,
            "policy_sha256": policy, "log": str(log), "log_sha256": sha(log.read_bytes()),
            "report": str(report), "report_sha256": sha(report.read_bytes()) if report.exists() else None}
    if findings:
        item["finding_metadata"] = [{"rule": f.get("RuleID"), "file": f.get("File"),
                                     "start_line": f.get("StartLine")} for f in findings]
    record.setdefault("secret_scans", []).append(item)
    save()
    assert result.returncode == 0 and isinstance(findings, list) and not findings, "Triage retained secret findings"


def materialize(tree, names, directory):
    root = Path(directory).resolve(strict=True)
    assert root.is_relative_to(EV) and root != EV
    blobs = []
    for name in sorted(names):
        entry = git("ls-tree", "-z", tree, "--", name).split(b"\0")
        assert len(entry) == 2 and not entry[1]
        metadata, actual = entry[0].split(b"\t", 1)
        mode, kind, oid = metadata.decode().split()
        assert actual.decode() == name and kind == "blob" and mode in {"100644", "100755"}
        raw = git("cat-file", "blob", oid)
        target = root / name
        assert target.resolve().is_relative_to(root) and not target.exists()
        target.parent.mkdir(parents=True, exist_ok=True)
        with target.open("xb") as stream:
            stream.write(raw)
        blobs.append({"path": name, "git_blob": oid, "sha256": sha(raw), "bytes": len(raw)})
    return blobs

        git("diff", "--cached", "--check", PARENT)
        assert not git("diff", "--name-only").strip()
        verify_files(x["physical"])
        verify_files([[f["path"], f["sha256"]] for f in validation["files"]])
        pending_source()
        assert set(git("diff", "--cached", "--name-only", PARENT).decode().splitlines()) == set(source["code_paths"]) | published
        record["status"] = "staged-and-verified"
        save()
        result = git("commit", "-m", "Add governed review corrections for published Factory results")
        with (EV / "issue95-publication-commit-r1.log").open("xb") as stream:
            stream.write(result)
        head = git("rev-parse", "HEAD").decode().strip()
        assert git("show", "-s", "--format=%P", head).decode().split() == PARENTS
        assert not Path(git("rev-parse", "--path-format=absolute", "--git-path", "MERGE_HEAD").decode().strip()).exists()
        assert git("rev-parse", "HEAD^{tree}").decode().strip() == validation["publication_tree"]
        assert not git("status", "--porcelain=v1", "--untracked-files=all").strip()
        assert set(git("diff", "--name-only", PARENT, head).decode().splitlines()) == set(source["code_paths"]) | published
        verify_files(x["physical"])
        verify_files([[f["path"], f["sha256"]] for f in validation["files"]])
        record.update(status="committed-and-verified", head=head, committed_tree=validation["publication_tree"],
                      commit_log_sha256=sha(result), worktree_clean=True, ordinary_two_parent_commit=True,
                      prerequisite_head_preserved_as_parent=True)
    except Exception as error:
        record.update(status="failed", failure=str(error), failure_type=type(error).__name__)
        raise
    finally:
        record["finished_at_utc"] = now()
        save()
        print(json.dumps({k: record.get(k) for k in ["status", "head", "tested_tree", "publication_tree", "failure"]}))


def scan_committed():
    receipt_path = EV / "issue95-publication-committed-scan-r1.json"
    assert not receipt_path.exists()
    committed = load(EV / "issue95-publication-commit-r1.json")
    assert committed["status"] == "committed-and-verified"
    head = committed["head"]
    source, validation = load(EV / "issue95-publication-source-r1.json"), load(validation_receipt(2))
    assert git("rev-parse", "HEAD").decode().strip() == head
    assert git("rev-parse", "HEAD^{tree}").decode().strip() == validation["publication_tree"]
    assert git("show", "-s", "--format=%P", head).decode().split() == PARENTS
    assert git("branch", "--show-current").decode().strip() == BRANCH
    assert not git("status", "--porcelain=v1", "--untracked-files=all").strip()
    verify_validation(validation)
    record = {"issue": 95, "status": "running", "started_at_utc": now(), "head": head, "main": MAIN,
              "parents": PARENTS, "tree": validation["publication_tree"], "secret_scans": [],
              "scope": "Every changed committed blob and all introduced history since main; merge diffs enabled."}

    def save():
        receipt_path.write_text(json.dumps(record, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")

    save()
    try:
        policy = policy_state()
        assert not git("diff", "--name-only", PARENT, head, "--", *policy).strip()
        published = {f["path"] for f in validation["files"]}
        names = set(git("diff", "--name-only", "--no-renames", PARENT, head).decode().splitlines())
        assert names == set(source["code_paths"]) | published
        with tempfile.TemporaryDirectory(prefix="issue95-committed-owned-blobs-", dir=EV) as directory:
            owned = Path(directory).resolve(strict=True)
            assert owned.parent == EV.resolve() and owned.name.startswith("issue95-committed-owned-blobs-")
            record["committed_blobs"] = materialize(head, names, owned)
            save()
            scan_secrets("changed-blobs", ["dir", str(owned)], record, save, "issue95-committed-scan-r1", policy)
        history = git("log", "-m", "--format=%H", "--name-only", MAIN + ".." + head)
        assert head.encode() in history and all(name.encode() in history for name in names)
        record.update(history_log_options="-m " + MAIN + ".." + head,
                      history_listing_sha256=sha(history))
        save()
        scan_secrets("introduced-history", ["git", str(ROOT), "--log-opts=" + record["history_log_options"]],
                     record, save, "issue95-committed-scan-r1", policy)
        assert git("rev-parse", "HEAD").decode().strip() == head and policy_state() == policy
        assert not git("status", "--porcelain=v1", "--untracked-files=all").strip()
        verify_files(source["physical_files"])
        verify_files([[f["path"], f["sha256"]] for f in validation["files"]])
        record.update(status="passed", exit_code=0, finding_count=0, worktree_unchanged=True)
    except Exception as error:
        record.update(status="failed", failure=str(error), failure_type=type(error).__name__)
        raise
    finally:
        record["finished_at_utc"] = now()
        save()
        print(json.dumps({k: record.get(k) for k in ["status", "head", "finding_count", "exit_code", "failure"]}))
