import hashlib

from pathlib import Path

EV = Path(__file__).resolve().parent


def derive(source, destination, transform):
    source, destination = EV / source, EV / destination
    assert not destination.exists(), f'Preserve existing evidence: {destination}'
    original = source.read_text(encoding='utf-8-sig')
    updated = transform(original)
    assert updated != original
    destination.write_text(updated, encoding='utf-8', newline='\n')
    return {'source': str(source), 'destination': str(destination),
            'sha256': hashlib.sha256(destination.read_bytes()).hexdigest()}


def replace_once(text, old, new):
    assert text.count(old) == 1, repr(old[:120])
    return text.replace(old, new)
