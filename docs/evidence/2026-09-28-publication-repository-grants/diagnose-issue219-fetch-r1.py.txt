import datetime
import json
import os
from pathlib import Path
import subprocess
import time

EV = Path(__file__).resolve().parent
ROOT = Path(r'<USERPROFILE>\code\qa\ecorp-issue219-fetch-diagnostic-r1')
REMOTE = Path(r'<USERPROFILE>\code\qa\ecorp-issue219-browser-r11\evidence\browser\target-remote.git')
RECEIPT = EV / 'issue219-fetch-diagnostic-r1.json'
assert not ROOT.exists() and not RECEIPT.exists()
ROOT.mkdir()
(ROOT / 'home').mkdir()
(ROOT / 'temp').mkdir()
(ROOT / 'empty.gitconfig').write_text('', encoding='utf-8')
env = {key: value for key, value in os.environ.items() if key.upper() in {
    'PATH', 'PATHEXT', 'SYSTEMROOT', 'WINDIR', 'COMSPEC', 'SYSTEMDRIVE',
    'PROCESSOR_ARCHITECTURE', 'NUMBER_OF_PROCESSORS', 'OS',
}}
env.update({'HOME': str(ROOT / 'home'), 'USERPROFILE': str(ROOT / 'home'),
            'APPDATA': str(ROOT / 'home'), 'LOCALAPPDATA': str(ROOT / 'home'),
            'TEMP': str(ROOT / 'temp'), 'TMP': str(ROOT / 'temp'),
            'GIT_CONFIG_GLOBAL': str(ROOT / 'empty.gitconfig'), 'GIT_CONFIG_NOSYSTEM': '1',
            'GIT_TERMINAL_PROMPT': '0'})
receipt = {'purpose': 'Diagnose retained browser r11 local Git fetch timeout without changing product limits or retained fixture',
           'status': 'running', 'owned_root': str(ROOT), 'retained_remote': str(REMOTE),
           'started_at_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'commands': []}

def save():
    RECEIPT.write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')

def run(label, args, timeout=150):
    child_env = dict(env)
    child_env['GIT_TRACE2_EVENT'] = str(ROOT / (label + '.trace.jsonl'))
    started = time.monotonic()
    with (ROOT / (label + '.stdout.log')).open('wb') as out, (ROOT / (label + '.stderr.log')).open('wb') as err:
        proc = subprocess.Popen(['git', *args], cwd=ROOT, env=child_env, stdout=out, stderr=err,
                                creationflags=subprocess.CREATE_NO_WINDOW)
        try:
            code = proc.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            subprocess.run(['taskkill.exe', '/PID', str(proc.pid), '/T', '/F'], stdout=subprocess.DEVNULL,
                           stderr=subprocess.DEVNULL, creationflags=subprocess.CREATE_NO_WINDOW)
            proc.wait(timeout=15)
            code = -1
    result = {'label': label, 'args': args, 'exit_code': code, 'elapsed_seconds': round(time.monotonic() - started, 3)}
    receipt['commands'].append(result)
    save()
    print(json.dumps(result), flush=True)
    assert code == 0, label
    return (ROOT / (label + '.stdout.log')).read_text(encoding='utf-8', errors='replace').strip()

save()
try:
    receipt['retained_refs_before'] = run('retained-refs-before', ['--git-dir', str(REMOTE), 'show-ref', '--head'])
    for name in ['first', 'second']:
        bare = ROOT / (name + '.git')
        bare.mkdir()
        run(name + '-init', ['-C', str(bare), 'init', '--bare'])
        run(name + '-origin', ['-C', str(bare), 'remote', 'add', 'origin', str(REMOTE)])
        run(name + '-fetch', ['-C', str(bare), 'fetch', '--no-tags', 'origin', 'HEAD'])
        receipt[name + '_fetched_head'] = run(name + '-head', ['-C', str(bare), 'rev-parse', 'FETCH_HEAD^{commit}'])
    receipt['retained_refs_after'] = run('retained-refs-after', ['--git-dir', str(REMOTE), 'show-ref', '--head'])
    assert receipt['retained_refs_before'] == receipt['retained_refs_after']
    assert receipt['first_fetched_head'] == receipt['second_fetched_head'] == '878a1774774b0630c904cbaf4b05e1b346777817'
    receipt['status'] = 'passed'
except Exception as error:
    receipt['status'] = 'failed'
    receipt['failure'] = str(error)
finally:
    receipt['finished_at_utc'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    save()
    env.clear()
print(json.dumps({'status': receipt['status'], 'receipt': str(RECEIPT)}), flush=True)
raise SystemExit(0 if receipt['status'] == 'passed' else 1)
