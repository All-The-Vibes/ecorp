async function request(url, init) {
  const response = await fetch(`${server}${url}`, init)
  const body = response.status === 204 ? null : await response.json()
  return { response, body }
}

async function requestOk(url, init) {
  const result = await request(url, init)
  if (!result.response.ok) {
    throw new Error(
      `${init?.method ?? 'GET'} ${url}: ${result.response.status} ${JSON.stringify(result.body)}`,
    )
  }
  return result.body
}

async function waitFor(demo, predicate, description, timeoutMs = 300_000) {
  const deadline = Date.now() + timeoutMs
  while (Date.now() < deadline) {
    const state = await snapshot(demo)
    const value = predicate(state)
    if (value) return { state, value }
    await new Promise((resolve) => setTimeout(resolve, 100))
  }
  throw new Error(`timed out waiting for ${description}`)
}

async function openBrowserClient(demo, actorId, afterSeq = 0) {
  const events = []
  let cursor = afterSeq
  let readyResolve
  let readyReject
  let closedResolve
  const ready = new Promise((resolve, reject) => {
    readyResolve = resolve
    readyReject = reject
  })
  const closed = new Promise((resolve) => {
    closedResolve = resolve
  })
  const socket = new WebSocket(
    `${socketBase}/ws/corps/${demo.corp_id}?actor_id=${actorId}&after_seq=${afterSeq}`,
  )
  const timeout = setTimeout(() => {
    socket.close()
    readyReject(new Error(`browser ${actorId} replay timed out`))
  }, 10_000)
  socket.onerror = () => {
    clearTimeout(timeout)
    readyReject(new Error(`browser ${actorId} websocket failed`))
  }
  socket.onclose = () => closedResolve()
  socket.onmessage = (message) => {
    const payload = JSON.parse(message.data)
    if (payload.type === 'event') {
      events.push(payload.event)
      cursor = Math.max(cursor, payload.event.seq)
    } else if (payload.type === 'ready') {
      clearTimeout(timeout)
      cursor = Math.max(cursor, payload.replayed_through)
      readyResolve()
    }
  }
  await ready
  return { actorId, events, socket, closed, cursor: () => cursor }
}

function assertUniqueReplay(events, afterSeq) {
  const sequences = events.map((event) => event.seq)
  assert.equal(new Set(sequences).size, sequences.length)
  assert.ok(sequences.every((sequence) => sequence > afterSeq))
  assert.ok(
    sequences.every(
      (sequence, index) => index === 0 || sequence > sequences[index - 1],
    ),
  )
}

// Fixture-only scheduling proxy: all responses come from the real native server.
// It never changes eligibility, lease timestamps, publication state or replies.
async function createQueueSchedulingProxy(demo, publicationId) {
  const gate = () => {
    let resolve
    const promise = new Promise(done => { resolve = done })
    return { promise, resolve }
  }
  const firstObserved = gate(), firstRelease = gate(), busyObserved = gate()
  const secondObserved = gate(), secondRelease = gate()
  const events = [], errors = []
  let queueReads = 0, closed = false
  const wait = async (signal, label) => {
    let timer
    try {
      return await Promise.race([signal.promise, new Promise((_, reject) => {
        timer = setTimeout(() => reject(new Error(`Scheduling proxy timed out: ${label}`)), 60000)
      })])
    } finally { clearTimeout(timer) }
  }
  const proxy = createServer((incoming, outgoing) => {
    const handle = async () => {
      const url = new URL(incoming.url, server)
      assert.equal(url.origin, server)
      assert.ok(url.pathname.startsWith(`/api/corps/${demo.corp_id}/factory/`))
      const queue = incoming.method === 'GET' && url.pathname.endsWith('/publisher/queue')
      const ordinal = queue ? ++queueReads : null
      if (ordinal === 2) {
        events.push({ event: 'second_queue_waiting_before_native_request', at: new Date().toISOString() })
        secondObserved.resolve()
        await secondRelease.promise
      }
      if (closed) { outgoing.writeHead(503); outgoing.end(); return }
      const chunks = []
      let length = 0
      for await (const chunk of incoming) {
        length += chunk.length
        assert.ok(length <= 4 * 1024 * 1024)
        chunks.push(chunk)
      }
      const headers = Object.fromEntries(Object.entries(incoming.headers).filter(([name]) =>
        !['host', 'connection', 'content-length', 'transfer-encoding'].includes(name)))
      const response = await fetch(url, {
        method: incoming.method, headers,
        ...(chunks.length ? { body: Buffer.concat(chunks) } : {}),
        redirect: 'error', signal: AbortSignal.timeout(60000),
      })
      const bytes = Buffer.from(await response.arrayBuffer())
      if (queue) {
        assert.equal(response.status, 200)
        const page = JSON.parse(bytes.toString('utf8'))
        assert.deepEqual(page.publication_ids, [publicationId])
        events.push({ event: 'native_queue_reply', ordinal, at: new Date().toISOString(), publication_id: publicationId })
        if (ordinal === 1) {
          firstObserved.resolve()
          await firstRelease.promise
        }
      }
      if (incoming.method === 'POST' && url.pathname.endsWith(`/publications/${publicationId}/claim`)) {
        assert.equal(response.status, 200)
        const reply = JSON.parse(bytes.toString('utf8'))
        const busy = reply.busy === true
        events.push({ event: 'native_claim_reply', busy, at: new Date().toISOString(), publication_id: reply.publication.id })
        if (busy) busyObserved.resolve()
      }
      outgoing.writeHead(response.status, {
        'content-type': response.headers.get('content-type') ?? 'application/octet-stream',
        'content-length': String(bytes.length),
      })
      outgoing.end(bytes)
    }
    void handle().catch(error => {
      errors.push(safeError(error))
      outgoing.destroy()
    })
  })
  await new Promise((resolve, reject) => {
    proxy.once('error', reject)
    proxy.listen(0, '127.0.0.1', resolve)
  })
  return {
    origin: `http://127.0.0.1:${proxy.address().port}`, events, errors,
    firstObserved: () => wait(firstObserved, 'eligible first native queue response'),
    busyObserved: () => wait(busyObserved, 'actual busy claim'),
    secondObserved: () => wait(secondObserved, 'same watcher second poll'),
    releaseFirst: () => { events.push({ event: 'first_queue_released_after_winner_claim', at: new Date().toISOString() }); firstRelease.resolve() },
    releaseSecond: () => { events.push({ event: 'second_queue_released_after_observed_expiry', at: new Date().toISOString() }); secondRelease.resolve() },
    releaseAll: () => { closed = true; firstRelease.resolve(); secondRelease.resolve() },
    close: async () => { proxy.closeAllConnections(); await new Promise(resolve => proxy.close(resolve)) },
  }
}

async function screenshot(label) {
  const originalViewport = page.viewportSize()
  try {
    for (const viewport of [{ width: 390, height: 844 }, { width: 1440, height: 1050 }]) {
      await page.setViewportSize(viewport)

      await card.waitFor({ state: 'visible' })
      await card.scrollIntoViewIfNeeded()
      const layout = await card.evaluate(element => {
        const rect = element.getBoundingClientRect()
        return { viewport_width: window.innerWidth, document_width: document.documentElement.scrollWidth,
          card_x: rect.x, card_right: rect.right, card_width: rect.width,
          card_client_width: element.clientWidth, card_scroll_width: element.scrollWidth }
      })
      assert.ok(layout.document_width <= viewport.width + 1, `Document overflows at ${viewport.width}px`)
      assert.ok(layout.card_x >= -1 && layout.card_right <= viewport.width + 1, `Publication card exceeds ${viewport.width}px viewport`)
      assert.ok(layout.card_scroll_width <= layout.card_client_width + 1, `Publication card overflows at ${viewport.width}px`)
      const name = `${label}-${viewport.width}px`
      const destination = path.join(artifactRoot, `${name}.png`)
      await page.screenshot({ path: destination, fullPage: true })
      report.screenshots.push({ label: name, viewport, layout, path: destination,
        sha256: crypto.createHash('sha256').update(await readFile(destination)).digest('hex') })
      await save()
    }
  } finally {
    await page.setViewportSize(originalViewport)
  }
}
