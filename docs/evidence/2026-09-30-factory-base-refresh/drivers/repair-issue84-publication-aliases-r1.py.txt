"""Preserve the failed packet and consistently alias synthetic decision IDs."""
from issue84_publication_lib_r2 import *
import ast

source = load(EV / "issue84-publication-source-r1.json")
assembly_path = EV / "issue84-publication-packet-r1.json"
assembly = load(assembly_path)
failed_path = EV / "issue84-publication-evidence-validation-r1.json"
failed = load(failed_path)
receipt_path = EV / "issue84-publication-alias-correction-r1.json"
next_assembly = EV / "issue84-publication-packet-r2.json"
snapshot = EV / "issue84-publication-packet-before-alias-r1"
assert not receipt_path.exists() and not next_assembly.exists() and not snapshot.exists()
assert failed["status"] == "failed" and failed["secret_scan"]["exit_code"] == 42
findings = failed["secret_scan"]["finding_metadata"]
assert len(findings) == 3 and all(f["rule"] == "generic-api-key" for f in findings)
expected_names = {
    "acceptance/acceptance.json",
    "history/acceptance-r8/acceptance.json",
    "history/acceptance-r9/acceptance.json",
}
assert {Path(f["file"]).relative_to(PACKET).as_posix() for f in findings} == expected_names
pending_source()
original_files(source["physical_files"])
initial_index = index_digest()
assert initial_index == source["original_index_sha256"] == assembly["original_index_sha256"]
assert sha((PACKET / "summary.json").read_bytes()) == assembly["summary_sha256"]
old_files = sorted(p.relative_to(PACKET).as_posix() for p in PACKET.rglob("*") if p.is_file())
assert old_files == assembly["publication_files"]
expected_hashes = {Path(f["path"]).relative_to(PACKET_REL).as_posix(): f["sha256"] for f in failed["files"]}
assert set(old_files) == set(expected_hashes)
for name in old_files:
    path = PACKET / name
    canonical_single_link_file(path)
    assert sha(path.read_bytes()) == expected_hashes[name], name

record = {
    "issue": 84, "status": "preparing", "started_at_utc": now(),
    "base": MAIN, "tested_tree": source["tested_tree"],
    "physical_files_sha256": PHYSICAL,
    "failed_validation_sha256": sha(failed_path.read_bytes()),
    "scope": "Evidence packaging only; original source, acceptance, counts and historical receipts remain unchanged.",
    "classification": "Three generic-api-key findings are synthetic request decision_key UUIDs captured from an automated browser fixture.",
    "semantic_source": "tools/e2e_factory_base_refresh.mjs:282-285 verifies the browser request decision_key is a UUID; it is an idempotency identifier, not a credential.",
    "policy_changed": False, "snapshot": str(snapshot), "changes": [],
}
def save():
    receipt_path.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
save()
try:
    snapshot.mkdir()
    for name in old_files:
        path, target = PACKET / name, snapshot / name
        target.parent.mkdir(parents=True, exist_ok=True)
        with target.open("xb") as stream:
            stream.write(path.read_bytes())
        assert sha(target.read_bytes()) == expected_hashes[name]
    record["preserved_files"] = [{"path": n, "sha256": expected_hashes[n]} for n in old_files]
    aliases, paths = {}, []
    def discover(value, location, name):
        if isinstance(value, dict):
            for key, child in value.items():
                if key == "decision_key":
                    assert isinstance(child, str) and re.fullmatch(r"[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}", child, re.I)
                    alias = aliases.setdefault(child, f"<decision-{len(aliases)+1:03d}>")
                    paths.append({"file": name, "json_path": location + "." + key, "alias": alias})
                else:
                    discover(child, location + "." + key, name)
        elif isinstance(value, list):
            for index, child in enumerate(value):
                discover(child, f"{location}[{index}]", name)
    for name in sorted(expected_names):
        discover(load(PACKET / name), "$", name)
    assert len(paths) == len(aliases) == 3
    for name in old_files:
        path = PACKET / name
        if path.suffix == ".png":
            continue
        raw = path.read_bytes()
        revised = raw
        count = 0
        for identifier, alias in aliases.items():
            count += revised.count(identifier.encode())
            revised = revised.replace(identifier.encode(), alias.encode())
        if revised != raw:
            assert name in expected_names, "Unexpected occurrence needs explicit triage"
            path.write_bytes(revised)
            record["changes"].append({"file": name, "before_sha256": sha(raw), "after_sha256": sha(revised), "replacements": count})
    assert len(record["changes"]) == 3
    record["aliased_json_paths"] = paths
    aliases.clear()
    summary = load(PACKET / "summary.json")
    for artifact in summary["artifacts"]:
        path = PACKET / artifact["file"]
        if sha(path.read_bytes()) != artifact["published_sha256"]:
            assert artifact["file"] in expected_names
            artifact["published_sha256"] = sha(path.read_bytes())
            artifact["bytes"] = path.stat().st_size
            artifact["transform"] += "; synthetic decision UUID consistently aliased after scanner triage"
    note = ("The first evidence validator exited one after three generic-api-key alerts for synthetic browser decision_key UUIDs. "
            "The original failed packet and scan receipt were preserved locally. Published copies consistently alias those fixture "
            "idempotency identifiers; no scanner policy or ignore rule changed. Product source, original acceptance, test counts "
            "and original artifact hashes are unchanged. The revised packet requires a fresh complete evidence validation.")
    summary["qualifications"].append(note)
    summary["redactions"].extend({"file": p["file"], "json_path": p["json_path"], "alias": p["alias"],
                                  "treatment": "Synthetic fixture decision identifier consistently aliased after native scanner triage"}
                                 for p in paths)
    personal = re.compile(r"(?:[a-z]:)?[/\\]+Users[/\\]+[^/\\\s\"'<>]+", re.I)
    def sanitized(raw):
        text = raw.decode("utf-8-sig")
        text = re.sub(r"(?i)C\^?:\^?<USERPROFILE>\\shyamsridhar", "<USERPROFILE>", text)
        return personal.sub("<USERPROFILE>", text).replace("\r\n", "\n").encode()
    def add_artifact(name, raw, source_name):
        target = PACKET / name
        assert not target.exists()
        target.parent.mkdir(parents=True, exist_ok=True)
        published = sanitized(raw)
        with target.open("xb") as stream:
            stream.write(published)
        summary["artifacts"].append({"file": name, "source_name": source_name, "original_sha256": sha(raw),
                                     "published_sha256": sha(published), "bytes": len(published),
                                     "transform": "UTF-8/LF and personal path redaction; no outcome or count changes"})
    add_artifact("history/publication-validation-r1-failed.json", failed_path.read_bytes(), failed_path.name)
    exit_path = EV / "issue84-publication-validation-r2-session-exit.json"
    add_artifact("history/publication-validation-r1-session-exit.json", exit_path.read_bytes(), exit_path.name)
    new_scripts = {}
    for name in ("validate", "commit", "scan"):
        old = EV / f"{name}-issue84-publication-r2.py"
        new = EV / f"{name}-issue84-publication-r3.py"
        assert not new.exists()
        text = old.read_text(encoding="utf-8")
        text = text.replace("issue84-publication-evidence-validation-r1.json", "issue84-publication-evidence-validation-r2.json")
        text = text.replace("issue84-publication-packet-r1.json", "issue84-publication-packet-r2.json")
        if name == "validate":
            text = text.replace("issue84-publication-validation-r1-", "issue84-publication-validation-r2-")
        ast.parse(text)
        with new.open("x", encoding="utf-8", newline="\n") as stream:
            stream.write(text)
        new_scripts[name] = new
        add_artifact("drivers/" + new.name + ".txt", new.read_bytes(), new.name)
    publisher = EV / "publish-issue84-r3.ps1"
    assert not publisher.exists()
    publisher_text = (EV / "publish-issue84-r2.ps1").read_text(encoding="utf-8")
    publisher_text = publisher_text.replace("issue84-publication-evidence-validation-r1.json", "issue84-publication-evidence-validation-r2.json")
    with publisher.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(publisher_text)
    add_artifact("drivers/" + Path(__file__).name + ".txt", Path(__file__).read_bytes(), Path(__file__).name)
    readme = PACKET / "README.md"
    text = readme.read_text(encoding="utf-8")
    readme.write_text(text + "\n" + note + "\n", encoding="utf-8", newline="\n")
    summary["publication_alias_correction"] = {
        "failed_validator_status": "failed", "scan_exit_code": 42, "finding_count": 3,
        "classification": record["classification"], "semantic_source": record["semantic_source"],
        "changes": record["changes"], "aliases": paths, "policy_changed": False,
        "original_receipts_preserved": True, "product_source_unchanged": True,
    }
    (PACKET / "summary.json").write_text(json.dumps(summary, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    original_files(source["physical_files"])
    pending_source()
    assert index_digest() == initial_index
    for name in old_files:
        assert sha((snapshot / name).read_bytes()) == expected_hashes[name]
    updated = dict(assembly)
    updated.update(recorded_at_utc=now(), supersedes_packet_receipt_sha256=sha(assembly_path.read_bytes()),
                   publication_files=sorted(p.relative_to(PACKET).as_posix() for p in PACKET.rglob("*") if p.is_file()),
                   summary_sha256=sha((PACKET / "summary.json").read_bytes()),
                   alias_correction_only=True, original_failed_packet_preserved=True)
    write(next_assembly, updated)
    record.update(status="corrected-packet-awaiting-validation", publication_files=len(updated["publication_files"]),
                  original_physical_files_unchanged=True, original_index_unchanged=True,
                  new_assembly_sha256=sha(next_assembly.read_bytes()))
finally:
    record["finished_at_utc"] = now()
    save()
print(json.dumps({k: record[k] for k in ("status", "tested_tree", "publication_files", "original_physical_files_unchanged", "original_index_unchanged")}))
