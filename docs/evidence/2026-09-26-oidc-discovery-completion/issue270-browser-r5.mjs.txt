function sanitized(error) {
  let text = String(error?.message || error)
  for (const user of users) text = text.split(user.token).join('[ephemeral token withheld]')
  return text
}
async function request(route, body, expected = 200, role = 0) {
  const headers = { 'content-type': 'application/json' }
  if (role !== null) headers.authorization = 'Bearer ' + (role === 'invalid' ? 'invalid-owned-fixture-token' : users[role].token)
  const response = await fetch(server + route, { method: body === undefined ? 'GET' : 'POST', headers,
    body: body === undefined ? undefined : JSON.stringify(body), redirect: 'error', signal: AbortSignal.timeout(15_000) })
  if (body !== undefined || response.status !== 200) requests.push({ method: body === undefined ? 'GET' : 'POST', route, status: response.status, expected, identity: role })
  assert.equal(response.status, expected, `Unexpected status for ${route}`)
  if (response.status === 204 || response.status >= 400) return null
  return response.json()
}

function view(state) {
  const mission = state.snapshot.missions.find(item => item.id === report.mission_id)
  assert.ok(mission, 'Owned mission missing')
  const tasks = state.snapshot.tasks.filter(item => item.mission_id === mission.id)
  assert.equal(tasks.length, 1)
  const task = tasks[0]
  return { mission, task, runs: state.snapshot.runs.filter(item => item.task_id === task.id), state }
}
function contractDigest(task) {
  const canonical = JSON.stringify({ contract: task.contract, policy: task.verification_policy }, (_, value) =>
    value && typeof value === 'object' && !Array.isArray(value) ? Object.fromEntries(Object.entries(value).sort(([a], [b]) => a.localeCompare(b))) : value)
  return createHash('sha256').update(canonical).digest('hex')
}
async function waitFor(predicate) {
  const deadline = Date.now() + 150_000
  while (Date.now() < deadline) {
    const current = view(await snapshot())
    assert.ok(!['failed', 'cancelled'].includes(current.mission.status), 'Owned mission failed')
    if (predicate(current)) return current
    await delay(250)
  }
  throw new Error('Owned mission did not reach its required persisted state')
}
