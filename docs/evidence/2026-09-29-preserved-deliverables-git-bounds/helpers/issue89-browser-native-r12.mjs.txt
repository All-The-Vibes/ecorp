async function git(cwd, ...args) {
  const result = await execFile('git', ['-C', cwd, ...args], {
    cwd: fixture, windowsHide: true, timeout: 120_000, maxBuffer: 8 * 1024 * 1024,
  })
  return result.stdout.trim()
}
async function request(route, body) {
  const url = new URL(route, server)
  assert.equal(url.origin, server)
  const response = await fetch(url, {
    ...(body === undefined ? {} : { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body) }),
    redirect: 'error', signal: AbortSignal.timeout(30_000),
  })
  const value = await response.json()
  assert.ok(response.ok, `${response.status} ${url.pathname}: ${JSON.stringify(value)}`)
  return value
}
const snapshot = () => request(`/api/corps/${demo.corp_id}/snapshot?actor_id=${demo.alice_actor_id}`)
async function waitFor(predicate, label, timeout = 180_000) {
  const deadline = Date.now() + timeout
  while (Date.now() < deadline) {
    const state = await snapshot()
    const value = predicate(state.snapshot, state)
    if (value) return { state: state.snapshot, value }
    await new Promise(resolve => setTimeout(resolve, 200))
  }
  const last = await snapshot()
  await artifact(`timeout-${label.replace(/[^a-z0-9]+/gi, '-')}`, last)
  throw new Error(`Timed out: ${label}`)
}
async function collectJournal() {
  eventSocket = new WebSocket(`${server.replace(/^http/, 'ws')}/ws/corps/${demo.corp_id}?actor_id=${demo.alice_actor_id}&after_seq=0`)
  await new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error('Native journal did not become ready')), 15_000)
    eventSocket.onerror = () => { clearTimeout(timer); reject(new Error('Native journal failed')) }
    eventSocket.onmessage = message => {
      const packet = JSON.parse(message.data)
      if (packet.type === 'event') events.set(packet.event.id, packet.event)
      if (packet.type === 'ready') { clearTimeout(timer); resolve() }
    }
  })
}
const journal = (runId, type) => [...events.values()].filter(event => event.aggregate_id === runId && (!type || event.type === type)).sort((a, b) => a.seq - b.seq)
async function capture(label, target = page) {

  await save()
}
async function ownedWorkspace(run) {
  const expected = path.join(fixture, 'runner-workspaces')
  const observed = await realpath(run.workspace_path)
  assert.ok(observed.toLowerCase().startsWith(expected.toLowerCase() + path.sep))
  assert.equal((await lstat(run.workspace_path)).isSymbolicLink(), false)
  assert.equal(await git(observed, 'rev-parse', 'HEAD'), BASE)
  assert.equal(await git(observed, 'branch', '--show-current'), run.workspace_branch)
  return observed
}

  const heads = (await git(imported, 'bundle', 'list-heads', bundlePath)).split('\n').map(line => line.trim().split(/\s+/))
  assert.equal(heads.length, 1)
  assert.equal(heads[0][0], source.head_commit)
  await git(imported, 'fetch', '--no-tags', bundlePath, heads[0][1])
  await git(imported, 'checkout', '--detach', source.head_commit)
  const exportedNames = (await git(imported, 'diff', '--name-only', BASE, 'HEAD')).split('\n')
  for (const name of [...appPaths, 'base.txt', 'resumed.txt', 'resume-prompt.txt']) assert.ok(exportedNames.includes(name))
  for (const [name, originalBytes] of Object.entries(sourceBytes)) assert.deepEqual(await readFile(path.join(imported, name)), originalBytes)
  const nativeTests = await execFile(process.execPath, ['--test', '--test-reporter=tap', `${APP}/app.test.mjs`], { cwd: imported, windowsHide: true, timeout: 30_000 })
  assert.match(nativeTests.stdout, /# tests 4/)
  assert.match(nativeTests.stdout, /# pass 4/)
  assert.match(nativeTests.stdout, /# fail 0/)
  await writeFile(path.join(output, 'exported-application-tests.log'), nativeTests.stdout + nativeTests.stderr, { flag: 'wx' })
  report.export = { artifact_id: source.artifact_id, sha256: source.sha256, bytes: source.bytes, head: source.head_commit,
    verified_tree: document.verified_tree, git_bundle_sha256: document.git_bundle_sha256, imported, changed_paths: exportedNames,
    original_application_bytes_identical: true, downloaded_headers_verified: true, independently_executed_application_tests: 4 }
  await capture('04-complete-source-accepted-and-exported')
  const publicFiles = new Map(await Promise.all(['index.html', 'app.mjs'].map(async name => [`/${name}`, await readFile(path.join(imported, APP, name))])))
  appServer = createServer((req, res) => {
    const name = req.url === '/' ? '/index.html' : req.url
    if (!['GET', 'HEAD'].includes(req.method) || !publicFiles.has(name)) { res.writeHead(404); res.end(); return }
    res.writeHead(200, { 'content-type': name.endsWith('.mjs') ? 'text/javascript; charset=utf-8' : 'text/html; charset=utf-8', 'x-content-type-options': 'nosniff' })
    res.end(req.method === 'HEAD' ? undefined : publicFiles.get(name))
  })
  await new Promise((resolve, reject) => { appServer.once('error', reject); appServer.listen(0, '127.0.0.1', resolve) })
  appOrigin = `http://127.0.0.1:${appServer.address().port}`
  report.stage = 'independently imported application browser flow'
  await save()
  appPage = await context.newPage()
  appPage.on('pageerror', error => report.page_errors.push(String(error)))
  await appPage.goto(appOrigin, { waitUntil: 'networkidle' })
  await appPage.getByRole('heading', { name: 'Incident desk', exact: true }).waitFor()
  await appPage.getByLabel('Incident title', { exact: true }).fill('Preserved application remains usable')
  // Native nested select labels include their option text. Match the inspected
  // label prefix and require an unambiguous control before interacting.
  assert.equal(await appPage.getByLabel(/^Priority/).count(), 1)
  assert.equal(await appPage.getByLabel(/^Show status/).count(), 1)
  await appPage.getByLabel(/^Priority/).selectOption('critical')
  await appPage.getByRole('button', { name: 'Add incident', exact: true }).click()
  await appPage.getByLabel('Search incidents', { exact: true }).fill('preserved application')
  await appPage.getByRole('button', { name: 'Resolve Preserved application remains usable', exact: true }).click()
  await appPage.getByLabel(/^Show status/).selectOption('resolved')
  await appPage.getByRole('button', { name: 'Reopen Preserved application remains usable', exact: true }).waitFor()
  await capture('05-exported-application-create-filter-resolve', appPage)
  await appPage.getByRole('button', { name: 'Reopen Preserved application remains usable', exact: true }).click()
  await appPage.getByLabel(/^Show status/).selectOption('open')
  await appPage.getByRole('button', { name: 'Resolve Preserved application remains usable', exact: true }).waitFor()
  await appPage.setViewportSize({ width: 390, height: 844 })
  assert.ok(await appPage.evaluate(() => document.documentElement.scrollWidth <= window.innerWidth + 1))
  await capture('06-exported-application-reopened-mobile', appPage)
  assert.equal(report.page_errors.length, 0)
  assert.equal(report.blocked_requests.length, 0)
  assert.equal(await git(sourceRepository, 'rev-parse', 'HEAD'), BASE)
  assert.equal(await git(sourceRepository, 'status', '--porcelain'), '')
  finalState = (await snapshot()).snapshot
  assert.equal(finalState.runs.filter(run => run.task_id === created.task_id).length, 2)
  await artifact('final-native-snapshot', finalState)
  report.corrected_run_id = corrected.id
  report.acceptance = {
    complete_application_preserved: true, narrowing_rejected_without_effects: true,
    full_scope_evidence_only_correction: true, same_native_session_and_worktree: true,
    fresh_verification_and_complete_bundle: true, exported_application_exercised: true,
    source_checkout_unchanged: true, provider_runs: 2,
  }
  report.status = 'passed'
} catch (error) {
  report.status = 'failed'
  report.failure = String(error.stack ?? error).slice(0, 8000)
  if (process.env.DATABASE_URL) report.failure = report.failure.split(process.env.DATABASE_URL).join('[private fixture credential]')
  if (page && !page.isClosed()) try { await capture('failure-current-browser') } catch {}
  if (appPage && !appPage.isClosed()) try { await capture('failure-exported-application', appPage) } catch {}
  if (demo) try { await artifact('failure-native-snapshot', (await snapshot()).snapshot) } catch {}
  process.exitCode = 1
} finally {
  if (eventSocket) eventSocket.close()
  if (browser) await browser.close()
  if (appServer) await new Promise(resolve => appServer.close(resolve))
  await artifact('native-journal', [...events.values()].sort((a, b) => a.seq - b.seq))
  report.finished_at_utc = new Date().toISOString()
  await save()
  console.log(JSON.stringify({ issue: 89, status: report.status, stage: report.stage, acceptance: report.acceptance, failure: report.failure }, null, 2))
}
