def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def load(path):

def write_json(path, value):

def file_map(entries):
    result = {entry["path"]: entry["sha256"] for entry in entries}

def count_rust(path):
    rows = re.findall(r"test result: (?:ok|FAILED)\. (\d+) passed; (\d+) failed; (\d+) ignored; (\d+) measured; (\d+) filtered out;",
                      Path(path).read_text(encoding="utf-8-sig"))
    assert rows, "No Rust result summaries in " + Path(path).name
    result = dict(zip(["passed", "failed", "ignored", "measured", "filtered_out"],
                      (sum(int(row[index]) for row in rows) for index in range(5))))
    result["summaries"] = len(rows)
    return result

personal = re.compile(r"(?:[a-z]:)?[/\\]+Users[/\\]+[^/\\\s\"'<>]+", re.I)


def redact(text):
    return personal.sub("<USERPROFILE>", text).replace("\r\n", "\n")


def sanitized(value):
    if isinstance(value, str):
        return redact(value)
    if isinstance(value, list):
        return [sanitized(item) for item in value]
    if isinstance(value, dict):
        result = {redact(key): sanitized(item) for key, item in value.items()}
        assert len(result) == len(value), "Redaction must not merge JSON keys"
        return result
    return value


def put(raw, name, original_name, binary=False):
    if binary:
        published = raw
        transformation = "none"
    elif name.endswith(".json"):
        published = (json.dumps(sanitized(json.loads(raw.decode("utf-8-sig"))), indent=2, ensure_ascii=False) + "\n").encode()
        transformation = "JSON parsed, profile roots redacted in string keys/values, reserialized as UTF-8/LF; original hash retained separately"
    else:
        published = redact(raw.decode("utf-8-sig")).encode()
        transformation = "UTF-8 without BOM, LF newlines, local profile roots replaced with <USERPROFILE>"
    target = PACKET / name
    assert target.resolve().is_relative_to(PACKET.resolve())
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("xb") as stream:
        stream.write(published)
    artifacts.append({"file": name, "source_name": original_name, "original_sha256": sha(raw),
                      "published_sha256": sha(published), "bytes": len(published), "transform": transformation})


def copy(path, name=None, expected=None, binary=False):
    path = Path(path)
    assert path.is_file() and not path.is_symlink()
    assert not any(word in path.name.lower() for word in ["credential", "password", "pgpass"])
    raw = path.read_bytes()
    if expected is not None:
        assert sha(raw) == expected, path.name
    put(raw, name or path.name, path.name, binary)
