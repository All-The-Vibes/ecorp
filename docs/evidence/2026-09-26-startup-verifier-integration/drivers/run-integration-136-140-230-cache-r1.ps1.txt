function Start-Owned([string]$Role,[string]$Binary,[string[]]$Arguments,[string]$Directory,[hashtable]$Environment) {
    $state.processes[$Role] = Start-LocalOwnedProcess -Role $Role -Workspace $qa -FilePath $Binary -ArgumentList $Arguments -WorkingDirectory $Directory -LogDirectory (Join-Path $qa 'logs') -Environment $Environment
    Save
}
function Stop-Owned([string]$Role) {
    if (!$state.processes[$Role]) { return }
    if (!(Get-LocalProcessIdentity -ProcessId $state.processes[$Role].pid)) { $receipt.cleanup += @(@{ role = $Role; result = 'already exited' }); return }
    if (!(Stop-LocalOwnedProcess -Record $state.processes[$Role] -Workspace $qa)) { throw "Ownership cannot authorize stopping $Role." }
    $receipt.cleanup += @(@{ role = $Role; result = 'exact owned process stopped' }); Save
}
function Check-Listener([string]$Role,[int]$Port) {
    if (!(Test-LocalOwnedProcess -Record $state.processes[$Role] -Workspace $qa)) { throw "Owned $Role is not running." }
    $listeners = @(Get-NetTCPConnection -State Listen -LocalPort $Port | Select-Object -ExpandProperty OwningProcess -Unique)
    if ($listeners.Count -ne 1 -or $listeners[0] -ne $state.processes[$Role].pid) { throw "Listener ownership mismatch for $Role." }
}
function Wait-Server([string]$Role) {
    $deadline = [DateTime]::UtcNow.AddSeconds(50)
    do {
        try { $health = Invoke-RestMethod -Uri "$server/health" -TimeoutSec 2; if ($health.mode -eq 'development') { Check-Listener $Role $ports.server; return } } catch {}
        if (!(Test-LocalOwnedProcess -Record $state.processes[$Role] -Workspace $qa)) { throw "$Role exited before readiness." }
        Start-Sleep -Milliseconds 200
    } while ([DateTime]::UtcNow -lt $deadline)
    throw 'Owned development server did not reach readiness.'
}
function Local-Post([string]$Route,[hashtable]$Body) {
    $response = Invoke-WebRequest -Uri ($server + $Route) -Method Post -ContentType 'application/json' -Body ($Body | ConvertTo-Json -Compress) -TimeoutSec 15 -SkipHttpErrorCheck
    if ($response.StatusCode -ne 200) { throw "Unexpected local response status for $Route." }
    $response.Content | ConvertFrom-Json
}
