function safeDiagnostic(value) {
  return String(value).slice(0, 4096)
    .replace(/\b(?:postgres(?:ql)?|https?):\/\/[^\s/@]+:[^\s/@]+@/giu, '[credential-bearing URL withheld]@')
    .replace(/\bBearer\s+\S+/giu, 'Bearer [withheld]')
    .replace(/\b((?:access[_-]?)?token|password|secret|credential|api[_-]?key)\b\s*[:=]\s*["']?[^\s,"'}]+/giu, '$1=[withheld]')
    .slice(0, 2048)
}
async function boundedError(response) {
  const reader = response.body.getReader()
  const pieces = []
  let size = 0, truncated = false
  try {
    while (size < 16384) {
      const { value, done } = await reader.read()
      if (done) break
      const remaining = 16384 - size
      pieces.push(value.subarray(0, remaining))
      size += Math.min(value.length, remaining)
      if (value.length >= remaining) { truncated = true; break }
    }
  } finally { await reader.cancel() }
  const bytes = Buffer.concat(pieces)
  const result = { bytes_retained_for_digest: bytes.length, truncated,
    bounded_body_sha256: createHash('sha256').update(bytes).digest('hex') }
  try {
    const parsed = JSON.parse(bytes.toString('utf8'))
    const message = typeof parsed.error === 'string' ? parsed.error :
      typeof parsed.message === 'string' ? parsed.message : parsed.error?.message
    if (typeof message === 'string') result.message = safeDiagnostic(message)
  } catch { result.message = 'Non-JSON error body withheld; bounded digest recorded' }
  return result
}
async function request(route, body, method = body === undefined ? 'GET' : 'POST') {
  assert.ok(route.startsWith('/api/'))
  const response = await fetch(server + route, { method, redirect: 'error',
    headers: body === undefined ? undefined : { 'content-type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body), signal: AbortSignal.timeout(10_000) })
  let detail
  if (response.status !== 200) detail = await boundedError(response)
  if (method !== 'GET' || response.status !== 200) {
    report.http_results.push({ route, method, status: response.status, error: detail })
    verify(method + ' ' + route + ': HTTP status', () =>
      assert.equal(response.status, 200, route + ': ' + (detail?.message ?? 'unexpected HTTP status')))
  }
  return response.json()
}
async function downloadProviderEvidence(event, current) {
  const artifact = event.payload
  assert.match(artifact.artifact_id, /^[0-9a-f-]{36}$/iu)
  const route = api('/artifacts/' + artifact.artifact_id + '?actor_id=' + demo.alice_actor_id)
  const response = await fetch(server + route, { redirect: 'error', signal: AbortSignal.timeout(10_000) })
  const detail = response.status === 200 ? undefined : await boundedError(response)
  report.http_results.push({ route, method: 'GET', status: response.status, error: detail })
  verify(current.id + ': authorized server download returns the exact provider artifact', () => {
    assert.equal(response.status, 200, detail?.message)
    assert.equal(response.headers.get('content-type'), 'application/json')
    assert.equal(response.headers.get('x-crony-artifact-role'), 'provider_evidence')
    assert.equal(response.headers.get('etag'), '"' + artifact.sha256 + '"')
    assert.ok(response.headers.get('x-crony-artifact-signature'))
    assert.ok(Number.isSafeInteger(artifact.bytes) && artifact.bytes > 0 && artifact.bytes <= 1024 * 1024)
    assert.equal(Number(response.headers.get('content-length')), artifact.bytes)
  })
  const bytes = Buffer.from(await response.arrayBuffer())
  verify(current.id + ': downloaded provider bytes match the authoritative artifact digest', () => {
    assert.equal(bytes.length, artifact.bytes)
    assert.equal(createHash('sha256').update(bytes).digest('hex'), artifact.sha256)
  })
  const destination = path.join(output, 'provider-artifact-' + current.id + '.json')
  await writeFile(destination, bytes, { flag: 'wx' })
  current.provider_artifact = { id: artifact.artifact_id, event_id: event.id, event_seq: event.seq,
    file_name: artifact.file_name, bytes: bytes.length, sha256: artifact.sha256, path: destination,
    server_role: 'provider_evidence', server_digest_verified: true }
  return JSON.parse(bytes)
}

const deadline = Date.now() + 10 * 60_000
async function waitFor(read, predicate, timeout = 90_000) {
  const until = Math.min(deadline, Date.now() + timeout)
  while (Date.now() < until) {
    const value = await read()
    if (predicate(value)) return value
    await delay(125)
  }
  throw new Error('Owned usage acceptance reached its observation ceiling')
}

  report.finished_at_utc = new Date().toISOString()
  report.counts = { passed: report.assertions.filter(item => item.passed).length,
    failed: report.assertions.filter(item => !item.passed).length,
    execution_errors: report.execution_errors.length,
    scenarios_failed: report.scenarios.filter(item => item.status === 'failed').length,
    scenarios_passed: report.scenarios.filter(item => item.status === 'passed').length,
    scenarios_unexecuted: corpus.cases.length - report.scenarios.length }
  await save()
  console.log(JSON.stringify({ status: report.status, counts: report.counts, failure: report.failure }))
  process.exitCode = report.status === 'passed' ? 0 : 1
}
