def sanitizer(text):
    wanted = []
    for node in ast.parse(text).body:
        if isinstance(node, ast.FunctionDef) and node.name in {"discover", "redact", "scrub"}:
            wanted.append(node)
        elif isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id in {"personal", "secret_key"} for t in node.targets):
            wanted.append(node)
    context = {"re": re, "secrets": set(), "identifiers": {}, "redactions": []}
    exec(compile(ast.Module(body=wanted, type_ignores=[]), "publication-sanitizer", "exec"), context)
    return context
