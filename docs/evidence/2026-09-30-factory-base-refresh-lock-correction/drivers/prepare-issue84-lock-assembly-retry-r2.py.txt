def sanitizer(text):
    parsed = ast.parse(text)
    wanted = []
    for node in parsed.body:
        if isinstance(node, ast.FunctionDef) and node.name in {"redact", "scrub"}:
            wanted.append(node)
        elif isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id in {"personal", "secret_key"} for t in node.targets):
            wanted.append(node)
    context = {"re": re, "secrets": set(), "identifiers": {}}
    exec(compile(ast.Module(body=wanted, type_ignores=[]), "publication-sanitizer", "exec"), context)
    return context
