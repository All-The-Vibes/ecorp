#requires -Version 7.5
param(
    [Parameter(Mandatory)][ValidatePattern('^r[0-9]+$')][string]$Revision,
    [Parameter(Mandatory)][string]$DatabaseReceipt,
    [Parameter(Mandatory)][string]$QaRoot,
    [int]$ServerPort = 19085,
    [int]$WebPort = 16085,
    [int]$DatabasePort = 59089
)
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $false
$product = '<USER_HOME>\.codex\worktrees\issue291-finite-budget\ecorp'
$qa = [IO.Path]::GetFullPath($QaRoot)
$pg = '<USER_HOME>\AppData\Local\Programs\ecorp-tools\postgresql-17.10\pgsql\bin'
$nodeDirectory = '<USER_HOME>\AppData\Local\Programs\ecorp-tools\node-v24.21.0-win-x64'
$playwright = '<USER_HOME>\.cache\codex-runtimes\codex-primary-runtime\dependencies\node\node_modules\playwright'
$python = '<USER_HOME>\.cache\codex-runtimes\codex-primary-runtime\dependencies\python\python.exe'
$node = Join-Path $nodeDirectory 'node.exe'
$prefix = "issue291-native-$Revision"
$receiptPath = Join-Path $PSScriptRoot "$prefix-lifecycle.json"
$buildReceiptPath = Join-Path $PSScriptRoot "$prefix-build.json"
$supervisor = '<USER_HOME>\code\ecorp\output\issue-completion\20260926T112626Z\issue224-stack-r1.ps1'
$snapshotter = Join-Path $PSScriptRoot 'issue291-source-snapshot-r1.mjs'
if ((Test-Path -LiteralPath $qa) -or (Test-Path -LiteralPath $receiptPath) -or (Test-Path -LiteralPath $buildReceiptPath)) { throw 'Preserve existing fixture and evidence.' }
if ([IO.Path]::GetDirectoryName($qa) -cne '<USER_HOME>\code\qa' -or [IO.Path]::GetFileName($qa) -notmatch '^issue224-planned-attempts-[a-zA-Z0-9-]+$') { throw 'Unexpected fixture path.' }
foreach ($name in @([Environment]::GetEnvironmentVariables('Process').Keys)) {
    if ($name -match '^(CRONY_|ECORP_|PG|GH_|GITHUB_|AZURE_|GIT_)' -or $name -in @('DATABASE_URL','OPENAI_API_KEY','ANTHROPIC_API_KEY','COPILOT_GITHUB_TOKEN','NODE_OPTIONS','npm_execpath','CARGO_TARGET_DIR','RUSTUP_TOOLCHAIN','RUSTFLAGS','RUSTDOCFLAGS')) { [Environment]::SetEnvironmentVariable($name,$null,'Process') }
}
$env:PATH = "$nodeDirectory;$pg;$(Split-Path -Parent $python);$env:PATH"
$env:PYTHON = $python
$env:CARGO_BUILD_JOBS = '2'
$env:CARGO_INCREMENTAL = '0'
$env:CARGO_PROFILE_DEV_DEBUG = '0'
$env:CARGO_PROFILE_TEST_DEBUG = '0'
$env:CARGO_NET_OFFLINE = 'true'
$env:RUST_TEST_THREADS = '1'
$receipt = [ordered]@{repository='All-The-Vibes/ecorp';issue=291;source=$product;status='running';wrapper_pid=$PID;started_at_utc=[DateTime]::UtcNow.ToString('o');qa_root=$qa;checks=@();failure=$null;cleanup='pending';scope='Prospective contributor regression. Native browser/server/PostgreSQL/runner with the existing synthetic Codex protocol peer. No real-provider, native Factory publication or independent human outcome acceptance.'}
function Save { [IO.File]::WriteAllText($receiptPath,($receipt | ConvertTo-Json -Depth 50)+"`n") }
function Check([string]$Name,[string]$Log,[int]$Code,[string]$Command) {
    $receipt.checks += @{name=$Name;command=$Command;exit_code=$Code;log=$Log;sha256=(Get-FileHash -LiteralPath $Log).Hash.ToLowerInvariant()}
    Save
    Write-Output "$Name exit=$Code"
    if ($Code) { throw "$Name failed; preserve original evidence." }
}
Save
Push-Location -LiteralPath $product
try {
    $before = & $node $snapshotter $product (Join-Path $PSScriptRoot "$prefix-source-before.json")
    if ($LASTEXITCODE) { throw 'Cannot bind source before native validation.' }
    $receipt.source_before = $before | ConvertFrom-Json -AsHashtable
    $db = Get-Content -LiteralPath $DatabaseReceipt -Raw | ConvertFrom-Json -AsHashtable -Depth 50
    if ($db.status -ne 'passed' -or !$db.source_unchanged -or $db.source_after.source_fingerprint -cne $receipt.source_before.source_fingerprint) { throw 'Require all five SQLx cases on the unchanged current source.' }
    $receipt.database_receipt = @{path=$DatabaseReceipt;sha256=(Get-FileHash -LiteralPath $DatabaseReceipt).Hash.ToLowerInvariant()}
    $receipt.supervisor = @{path=$supervisor;sha256=(Get-FileHash -LiteralPath $supervisor).Hash.ToLowerInvariant()}
    Save
    $numericLog = Join-Path $PSScriptRoot "$prefix-numeric-helper.log"
    & $node tools/test_token_ceiling.mjs *> $numericLog
    Check 'actual-numeric-regression-helper' $numericLog $LASTEXITCODE 'node tools/test_token_ceiling.mjs'
    $numericResult = Get-Content -LiteralPath $numericLog -Raw | ConvertFrom-Json -AsHashtable
    $numeric = Get-Content -LiteralPath $numericResult.receipt -Raw | ConvertFrom-Json -AsHashtable -Depth 30
    if ($numeric.status -ne 'passed' -or !$numeric.source_unchanged -or !$numeric.primary_lock_unchanged -or $numeric.source_before -cne $receipt.source_before.source_fingerprint) { throw 'Numeric helper did not validate this unchanged source.' }
    $receipt.numeric_receipt = @{path=$numericResult.receipt;sha256=(Get-FileHash -LiteralPath $numericResult.receipt).Hash.ToLowerInvariant();commands=$numeric.commands}
    Save
    $cargoJson = Join-Path $PSScriptRoot "$prefix-build.stdout.jsonl"
    $cargoStderr = Join-Path $PSScriptRoot "$prefix-build.stderr.log"
    & cargo build --locked --offline --message-format=json -p crony-server -p crony-runner 1> $cargoJson 2> $cargoStderr
    $buildCode = $LASTEXITCODE
    Check 'actual-native-binary-build' $cargoStderr $buildCode 'cargo build --locked --offline --message-format=json -p crony-server -p crony-runner'
    $messages = @(Get-Content -LiteralPath $cargoJson | ForEach-Object { $_ | ConvertFrom-Json -AsHashtable -Depth 50 })
    if ($messages[-1].reason -ne 'build-finished' -or !$messages[-1].success) { throw 'Missing successful Cargo build-finished record.' }
    $binaries = @()
    foreach ($role in @('server','runner')) {
        $artifact = @($messages | Where-Object { $_.reason -eq 'compiler-artifact' -and $_.target.name -eq "crony-$role" -and $_.target.kind -contains 'bin' -and $_.profile.test -eq $false -and $_.executable -and $_.manifest_path -ieq (Join-Path $product "crates/crony-$role/Cargo.toml") })
        if ($artifact.Count -ne 1) { throw "Ambiguous crony-$role compiler artifact." }
        $binary = (Resolve-Path -LiteralPath $artifact[0].executable).Path
        if ($binary -ine (Join-Path $product "target/debug/crony-$role.exe")) { throw 'Unexpected native executable path.' }
        $binaries += @{path=$binary;sha256=(Get-FileHash -LiteralPath $binary).Hash.ToLowerInvariant()}
    }
    $builtText = & $node $snapshotter $product (Join-Path $PSScriptRoot "$prefix-source-built.json")
    if ($LASTEXITCODE) { throw 'Cannot bind actual native build source.' }
    $built = $builtText | ConvertFrom-Json -AsHashtable
    if ($built.source_fingerprint -cne $receipt.source_before.source_fingerprint) { throw 'Source changed during native build.' }
    $build = [ordered]@{status='passed';source=$product;source_before=$receipt.source_before.source_fingerprint;source_after=$built.source_fingerprint;cargo_json=$cargoJson;cargo_json_sha256=(Get-FileHash -LiteralPath $cargoJson).Hash.ToLowerInvariant();binaries=$binaries;command='cargo build --locked --offline --message-format=json -p crony-server -p crony-runner';exit_code=$buildCode;finished_at_utc=[DateTime]::UtcNow.ToString('o')}
    [IO.File]::WriteAllText($buildReceiptPath,($build | ConvertTo-Json -Depth 15)+"`n")
    $receipt.build_receipt = @{path=$buildReceiptPath;sha256=(Get-FileHash -LiteralPath $buildReceiptPath).Hash.ToLowerInvariant()}
    Save
    $startLog = Join-Path $PSScriptRoot "$prefix-stack-start.log"
    & $supervisor -Phase Start -Repository $product -QaRoot $qa -PostgresBin $pg -ServerPort $ServerPort -WebPort $WebPort -DatabasePort $DatabasePort *> $startLog
    & $supervisor -Phase Status -Repository $product -QaRoot $qa -PostgresBin $pg *> (Join-Path $PSScriptRoot "$prefix-stack-status.log")
    $receipt.stack_started = @{log=$startLog;sha256=(Get-FileHash -LiteralPath $startLog).Hash.ToLowerInvariant()}
    Save
    $env:CRONY_TOKEN_CEILING_TEST = '1'
    $env:CRONY_PLAYWRIGHT_MODULE = $playwright
    $browserLog = Join-Path $PSScriptRoot "$prefix-browser.log"
    & $node tools/e2e_token_ceiling.mjs $qa $buildReceiptPath *> $browserLog
    $browserCode = $LASTEXITCODE
    try {
        $result = Get-Content -LiteralPath $browserLog -Raw | ConvertFrom-Json -AsHashtable -Depth 50
        if ($result.receipt) {
            $resultPath = [IO.Path]::GetFullPath($result.receipt)
            if (!$resultPath.StartsWith((Join-Path $qa 'evidence') + '\',[StringComparison]::OrdinalIgnoreCase)) { throw 'Browser receipt escaped the owned fixture.' }
            $receipt.browser_receipt = @{path=$resultPath;sha256=(Get-FileHash -LiteralPath $resultPath).Hash.ToLowerInvariant();status=$result.status}
        }
    } catch { $receipt.browser_result_parse_error=$_.Exception.Message }
    Check 'owned-browser-to-server-to-runner-finite-budget-regression' $browserLog $browserCode 'CRONY_TOKEN_CEILING_TEST=1 node tools/e2e_token_ceiling.mjs <owned-fixture> <actual-build-receipt>'
    if (!$receipt.browser_receipt) { throw 'Missing browser regression receipt.' }
    $browserReport = Get-Content -LiteralPath $receipt.browser_receipt.path -Raw | ConvertFrom-Json -AsHashtable -Depth 60
    if ($browserReport.status -ne 'passed' -or $browserReport.source_before -cne $receipt.source_before.source_fingerprint -or $browserReport.source_after -cne $receipt.source_before.source_fingerprint -or $browserReport.cases.Count -ne 18) { throw 'Incomplete exact-source browser regression.' }
    $receipt.browser_cases = $browserReport.cases
    $receipt.status = 'passed'
} catch { $receipt.status='failed';$receipt.failure=($_.Exception.Message -replace '\bpostgres(?:ql)?://\S+','[database URL withheld]') } finally {
    try {
        if (Test-Path -LiteralPath (Join-Path $qa 'ownership.json')) {
            & $supervisor -Phase Stop -Repository $product -QaRoot $qa -PostgresBin $pg *> (Join-Path $PSScriptRoot "$prefix-stack-stop.log")
            Import-Module (Join-Path $product 'tools/local_stack.psm1') -Force -DisableNameChecking
            $stopped = Read-LocalStackState -Path (Join-Path $qa 'ownership.json') -Workspace $qa
            foreach ($role in @('web','runner','server','postgres')) {
                if ($stopped.processes[$role] -and (Test-LocalOwnedProcess -Record $stopped.processes[$role] -Workspace $qa)) { throw "Owned $role process remains active." }
            }
            $receipt.cleanup = 'All exact owned stack processes verified stopped; source, database, worktrees, private credential files and evidence preserved.'
        } else { $receipt.cleanup = 'No stack ownership established; any files preserved.' }
    } catch { $receipt.status='failed';$receipt.cleanup_error=$_.Exception.Message;$receipt.cleanup='Preserve fixture: cleanup unverified.' }
    try {
        $afterText = & $node $snapshotter $product (Join-Path $PSScriptRoot "$prefix-source-after.json")
        if ($LASTEXITCODE) { throw 'Cannot bind source after native validation.' }
        $receipt.source_after = $afterText | ConvertFrom-Json -AsHashtable
        $receipt.source_unchanged = $receipt.source_after.source_fingerprint -ceq $receipt.source_before.source_fingerprint
        if (!$receipt.source_unchanged) { throw 'Native validation source changed.' }
    } catch { $receipt.status='failed';$receipt.source_failure=$_.Exception.Message }
    foreach ($name in @('CRONY_TOKEN_CEILING_TEST','CRONY_PLAYWRIGHT_MODULE')) { [Environment]::SetEnvironmentVariable($name,$null,'Process') }
    Pop-Location
    $receipt.finished_at_utc = [DateTime]::UtcNow.ToString('o')
    Save
}
[pscustomobject]$receipt | Select-Object issue,status,source_unchanged,cleanup,failure | ConvertTo-Json
if ($receipt.status -ne 'passed') { exit 1 }
