+import {
+  TITLE_LIMIT,
+  addTask,
+  createState,
+  deleteTask,
+  filterTasks,
+  parseState,
+  serializeState,
+  toggleTask
+} from "./tasks.mjs";
+
+const STORAGE_KEY = "focus-ledger.tasks";
+
+const elements = {
+  input: document.querySelector("#task-title"),
+  addButton: document.querySelector("#add-task"),
+  filter: document.querySelector("#show-tasks"),
+  list: document.querySelector("#task-list"),
+  empty: document.querySelector("#empty-state"),
+  error: document.querySelector("#form-error"),
+  status: document.querySelector("#storage-status")
+};
+
+let state = loadState();
+
+function loadState() {
+  try {
+    return parseState(window.localStorage.getItem(STORAGE_KEY));
+  } catch {
+    window.localStorage.removeItem(STORAGE_KEY);
+    elements.status.textContent = "Saved tasks were corrupted and have been reset.";
+    return createState();
+  }
+}
+
+function saveState() {
+  window.localStorage.setItem(STORAGE_KEY, serializeState(state));
+}
+
+function setError(message = "") {
+  elements.error.textContent = message;
+}
+
+function setStatus(message = "") {
+  elements.status.textContent = message;
+}
+
+function render() {
+  const visibleTasks = filterTasks(state, elements.filter.value);
+  elements.list.innerHTML = "";
+
+  if (visibleTasks.length === 0) {
+    elements.empty.hidden = false;
+    return;
+  }
+
+  elements.empty.hidden = true;
+
+  for (const task of visibleTasks) {
+    const item = document.createElement("li");
+    item.className = "task-item";
+    item.dataset.completed = String(task.completed);
+
+    const label = document.createElement("label");
+    label.className = "task-label";
+
+    const checkbox = document.createElement("input");
+    checkbox.type = "checkbox";
+    checkbox.checked = task.completed;
+    checkbox.setAttribute("aria-label", `Complete ${task.title}`);
+    checkbox.addEventListener("change", () => {
+      state = toggleTask(state, task.id);
+      saveState();
+      setStatus("");
+      render();
+    });
+
+    const text = document.createElement("span");
+    text.textContent = task.title;
+
+    label.append(checkbox, text);
+
+    const deleteButton = document.createElement("button");
+    deleteButton.type = "button";
+    deleteButton.className = "secondary";
+    deleteButton.textContent = "Delete";
+    deleteButton.setAttribute("aria-label", `Delete ${task.title}`);
+    deleteButton.addEventListener("click", () => {
+      state = deleteTask(state, task.id);
+      saveState();
+      setStatus("");
+      render();
+    });
+
+    item.append(label, deleteButton);
+    elements.list.append(item);
+  }
+}
+
+function handleAdd() {
+  setError("");
+  setStatus("");
+
+  try {
+    state = addTask(state, elements.input.value);
+    saveState();
+    elements.input.value = "";
+    render();
+    elements.input.focus();
+  } catch (error) {
+    const message = error instanceof Error ? error.message : `Task title must be ${TITLE_LIMIT} characters or fewer.`;
+    setError(message);
+  }
+}
+
+elements.addButton.addEventListener("click", handleAdd);
+elements.input.addEventListener("keydown", (event) => {
+  if (event.key === "Enter") {
+    event.preventDefault();
+    handleAdd();
+  }
+});
+elements.filter.addEventListener("change", () => render());
+
+render();

+const TITLE_LIMIT = 80;
+let nextId = 0;
+
+function assertTitle(title) {
+  const normalized = typeof title === "string" ? title.trim() : "";
+
+  if (!normalized) {
+    throw new Error("Task title is required.");
+  }
+
+  if (normalized.length > TITLE_LIMIT) {
+    throw new Error(`Task title must be ${TITLE_LIMIT} characters or fewer.`);
+  }
+
+  return normalized;
+}
+
+export function createState(tasks = []) {
+  return { tasks: [...tasks] };
+}
+
+export function addTask(state, title) {
+  const normalizedTitle = assertTitle(title);
+  const nextTask = {
+    id: globalThis.crypto?.randomUUID?.() ?? `task-${++nextId}`,
+    title: normalizedTitle,
+    completed: false
+  };
+
+  return {
+    tasks: [...state.tasks, nextTask]
+  };
+}
+
+export function toggleTask(state, id) {
+  let found = false;
+
+  const tasks = state.tasks.map((task) => {
+    if (task.id !== id) {
+      return task;
+    }
+
+    found = true;
+    return {
+      ...task,
+      completed: !task.completed
+    };
+  });
+
+  if (!found) {
+    throw new Error("Task not found.");
+  }
+
+  return { tasks };
+}
+
+export function deleteTask(state, id) {
+  const tasks = state.tasks.filter((task) => task.id !== id);
+
+  if (tasks.length === state.tasks.length) {
+    throw new Error("Task not found.");
+  }
+
+  return { tasks };
+}
+
+export function filterTasks(state, filter = "all") {
+  if (filter === "active") {
+    return state.tasks.filter((task) => !task.completed);
+  }
+
+  if (filter === "completed") {
+    return state.tasks.filter((task) => task.completed);
+  }
+
+  return [...state.tasks];
+}
+
+export function serializeState(state) {
+  return JSON.stringify(state.tasks);
+}
+
+export function parseState(serialized) {
+  if (serialized === null || serialized === "") {
+    return createState();
+  }
+
+  const parsed = JSON.parse(serialized);
+
+  if (!Array.isArray(parsed)) {
+    throw new Error("Saved tasks are invalid.");
+  }
+
+  return {
+    tasks: parsed.map((task) => {
+      if (
+        !task ||
+        typeof task.id !== "string" ||
+        typeof task.title !== "string" ||
+        typeof task.completed !== "boolean"
+      ) {
+        throw new Error("Saved tasks are invalid.");
+      }
+
+      return {
+        id: task.id,
+        title: assertTitle(task.title),
+        completed: task.completed
+      };
+    })
+  };
+}
+
+export { TITLE_LIMIT };

+import test from "node:test";
+import assert from "node:assert/strict";
+
+import {
+  TITLE_LIMIT,
+  addTask,
+  createState,
+  deleteTask,
+  filterTasks,
+  parseState,
+  serializeState,
+  toggleTask
+} from "../tasks.mjs";
+
+test("addTask trims title and stores active task", () => {
+  const state = addTask(createState(), "  Deep work block  ");
+
+  assert.equal(state.tasks.length, 1);
+  assert.equal(state.tasks[0].title, "Deep work block");
+  assert.equal(state.tasks[0].completed, false);
+});
+
+test("addTask rejects blank titles", () => {
+  assert.throws(() => addTask(createState(), "   "), {
+    message: "Task title is required."
+  });
+});
+
+test("addTask rejects titles longer than 80 characters", () => {
+  assert.throws(() => addTask(createState(), "x".repeat(TITLE_LIMIT + 1)), {
+    message: "Task title must be 80 characters or fewer."
+  });
+});
+
+test("toggleTask flips completion state", () => {
+  const initial = addTask(createState(), "Review notes");
+  const toggled = toggleTask(initial, initial.tasks[0].id);
+
+  assert.equal(toggled.tasks[0].completed, true);
+});
+
+test("deleteTask removes the matching task", () => {
+  const state = addTask(addTask(createState(), "Plan"), "Ship");
+  const remaining = deleteTask(state, state.tasks[0].id);
+
+  assert.deepEqual(remaining.tasks.map((task) => task.title), ["Ship"]);
+});
+
+test("filterTasks returns active and completed views", () => {
+  const first = addTask(createState(), "First");
+  const second = addTask(first, "Second");
+  const completed = toggleTask(second, second.tasks[1].id);
+
+  assert.deepEqual(filterTasks(completed, "active").map((task) => task.title), ["First"]);
+  assert.deepEqual(filterTasks(completed, "completed").map((task) => task.title), ["Second"]);
+});
+
+test("serializeState and parseState round-trip valid tasks", () => {
+  const added = addTask(createState(), "Archive");
+  const completed = toggleTask(added, added.tasks[0].id);
+  const parsed = parseState(serializeState(completed));
+
+  assert.equal(parsed.tasks.length, 1);
+  assert.equal(parsed.tasks[0].title, "Archive");
+  assert.equal(parsed.tasks[0].completed, true);
+});
+
+test("parseState rejects invalid saved data", () => {
+  assert.throws(() => parseState('{"bad":true}'), {
+    message: "Saved tasks are invalid."
+  });
+});
