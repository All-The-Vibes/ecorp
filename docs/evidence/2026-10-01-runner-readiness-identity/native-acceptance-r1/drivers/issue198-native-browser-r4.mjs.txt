async function request(route, body, expected = 200) {
  const response = await fetch(`${server}${route}`, {
    method: body === undefined ? 'GET' : 'POST',
    headers: { 'content-type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body),
    redirect: 'error', signal: AbortSignal.timeout(15000),
  })
  assert.equal(response.status, expected, `${route}: ${response.status} ${(response.status === expected ? '' : await response.clone().text()).slice(0, 1000)}`)
  return expected === 404 || expected === 204 ? null : response.json()
}

async function waitFor(fn, label, timeout = 60000) {
  const end = Date.now() + timeout
  while (Date.now() < end) {
    const value = await fn()
    if (value) return value
    await delay(350)
  }
  throw new Error(`Deadline exceeded: ${label}`)
}
async function writeAtomic(filename, value) {
  const temporary = path.join(fixture.output, `${filename}.tmp`)
  await writeFile(temporary, `${JSON.stringify(value, null, 2)}\n`)
  for (let attempt = 0; ; attempt++) {
    try { await rename(temporary, path.join(fixture.output, filename)); break }
    catch (error) {
      if (!['EPERM', 'EBUSY'].includes(error.code) || attempt >= 19) throw error
      await delay(50)
    }
  }
}

}

async function finished(operation) {
  const receipt = await waitFor(async () => {
    const next = await request(api(`/setup-operations/${operation.id}?actor_id=${demo.alice_actor_id}`))
    return ['queued', 'running'].includes(next.status) ? null : next

  report.operations.push(receipt)
  verify(`${receipt.id}: native receipt retains actor, Corp, runner and connection`, () => {
    assert.equal(receipt.corp_id, demo.corp_id)
    assert.equal(receipt.actor_id, demo.alice_actor_id)

    assert.equal(receipt.connection_id, operation.connection_id)
    assert.ok(receipt.report && receipt.report.connection_status)
  })
  await save()
  return receipt
}

async function runnerControl(action) {
  const nonce = `${action}-${randomUUID()}`
  await writeAtomic('runner-control.json', { nonce, action, runner_id: fixture.runner_id })
  return waitFor(() => readOptional(`runner-control-${nonce}.json`), `owned runner ${action}`)
}
async function permission(approval) {
  const fingerprint = hash(JSON.stringify(approval))
  await writeAtomic('pending-approval.json', { run_id: report.run_id, fingerprint, approval })
  await phase(`permission-review-${approval.id}`)
  const decision = await waitFor(async () => {
    const next = await readOptional(`decision-${approval.id}.json`)
    return next && next.approval_id === approval.id && next.fingerprint === fingerprint && typeof next.approve === 'boolean' ? next : null
  }, 'exact scoped command review by acceptance operator', 180000)
  const item = page.locator('.mission-approval-item').filter({ has: page.getByText(approval.action, { exact: true }) })
  const responsePending = page.waitForResponse(response => response.request().method() === 'POST' && response.url().includes(approval.id))
  await item.getByRole('button', { name: decision.approve ? 'Approve action' : 'Reject', exact: true }).click()
  const response = await responsePending
  assert.equal(response.status(), 200)
  report.permission_decisions.push({ approval, fingerprint, decision, response: await response.json(), decided_at: new Date().toISOString() })
  await save()
}
