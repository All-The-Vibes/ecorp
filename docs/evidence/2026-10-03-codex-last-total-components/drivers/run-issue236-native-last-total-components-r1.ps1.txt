    [Parameter(Mandatory)][ValidatePattern('^[0-9a-f]{64}$')][string]$ExpectedFingerprint,
    [Parameter(Mandatory)][string]$ReferenceManifest)
$Phase = 'green'
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $false
$product = '<candidate-checkout>'
$node = '<pinned-toolchain>\node-v24.21.0-win-x64\node.exe'
$qa = "<owned-qa-root>\issue236-native-usage-last-total-components-20261003-$Phase-r1"
$pg = '<pinned-toolchain>\postgresql-17.10\pgsql\bin'
$port = 59418
$dbUser = 'issue236'
$prefix = "issue236-native-last-total-components-$Phase-r1"
$receiptPath = Join-Path $PSScriptRoot "$prefix.json"
if ((Test-Path -LiteralPath $qa) -or (Test-Path -LiteralPath $receiptPath)) { throw 'Preserve existing fixture and receipts.' }
if (@(Get-NetTCPConnection -State Listen | Where-Object LocalPort -eq $port).Count) { throw 'Fixture port occupied; nothing stopped.' }
foreach ($name in @([Environment]::GetEnvironmentVariables('Process').Keys)) {
    if ($name -match '^(CRONY_|ECORP_|PG|GH_|GITHUB_|AZURE_|GIT_)' -or $name -in @('DATABASE_URL','OPENAI_API_KEY','ANTHROPIC_API_KEY','COPILOT_GITHUB_TOKEN','NODE_OPTIONS','CARGO_TARGET_DIR','RUSTUP_TOOLCHAIN','RUSTFLAGS','RUSTDOCFLAGS')) { [Environment]::SetEnvironmentVariable($name,$null,'Process') }
}
$env:PATH = $pg + ';<pinned-toolchain>\node-v24.21.0-win-x64;' + $env:PATH
$env:CARGO_TARGET_DIR = Join-Path $product 'target/issue236-review-r5'
if (!(Test-Path -LiteralPath $env:CARGO_TARGET_DIR)) { throw 'Fresh runner regression target is missing.' }
$env:CARGO_BUILD_JOBS = '2'
$env:CARGO_INCREMENTAL = '0'
$env:CARGO_PROFILE_DEV_DEBUG = '0'
$env:CARGO_PROFILE_TEST_DEBUG = '0'
$env:CARGO_NET_OFFLINE = 'true'
$env:RUST_TEST_THREADS = '1'
Import-Module (Join-Path $product 'tools/local_stack.psm1') -Force -DisableNameChecking
$beforeText = & $node (Join-Path $PSScriptRoot 'issue236-source-snapshot-r1.mjs') $product (Join-Path $PSScriptRoot "$prefix-source-before.json")
if ($LASTEXITCODE) { throw 'Cannot establish the native database validation source.' }
$before = $beforeText | ConvertFrom-Json -AsHashtable
if ($before.head -cne $ExpectedHead -or $before.branch -cne 'codex/issue236-usage-provenance' -or $before.source_fingerprint -cne $ExpectedFingerprint) { throw 'Native validation source differs from the reviewed correction.' }
$reference = Get-Content -LiteralPath $ReferenceManifest -Raw | ConvertFrom-Json -AsHashtable -Depth 60
$referenceHash = (Get-FileHash -LiteralPath $ReferenceManifest).Hash.ToLowerInvariant()
if ($reference.source -cne $product -or $reference.branch -cne 'codex/issue236-usage-provenance' -or
    $reference.head -cne $ExpectedHead -or $reference.source_fingerprint -cne $ExpectedFingerprint -or
    $before.source_fingerprint -cne $reference.source_fingerprint -or $before.status -cne $reference.status) {
    throw 'The complete physical source and Git status must match the reviewed manifest.'
}
New-Item -ItemType Directory -Path $qa | Out-Null
$sid = [Security.Principal.WindowsIdentity]::GetCurrent().User.Value
& icacls.exe $qa /inheritance:r /grant:r "*${sid}:(OI)(CI)F" '*S-1-5-18:(OI)(CI)F' *> $null
if ($LASTEXITCODE) { throw 'Cannot protect the fixture.' }
foreach ($dir in @('logs','credentials')) { New-Item -ItemType Directory -Path (Join-Path $qa $dir) | Out-Null }
$receipt = [ordered]@{
    repository='All-The-Vibes/ecorp'; issue=236; pr=402; review_comment_id=4175209381; phase=$Phase; related_issues=@(56,87,261); status='running';wrapper_pid=$PID;
    purpose='Source-bound usage and hard-control SQL regressions on the native last.totalTokens consistency correction. Includes persisted latest-total evidence, conflicting replays, overflow, unknown quantities and exactly-once accepted charges. Run both full discovered lanes. Fresh owned PostgreSQL usage presence, units, exact-assignment identity, replay, aliases, overflow, redaction, hard-control and revoked-read regressions. Synthetic accounting observations only; no budget grant, auditor execution, stopped experiment, real-provider, browser or hosted acceptance claimed.';
    source=$product; source_before=$before;reference_manifest=$ReferenceManifest;reference_manifest_sha256=$referenceHash; source_after=$null; source_unchanged=$null;
    qa_root=$qa; started_at_utc=[DateTime]::UtcNow.ToString('o');
    environment=@{CARGO_TARGET_DIR=$env:CARGO_TARGET_DIR;CARGO_BUILD_JOBS='2';CARGO_INCREMENTAL='0';CARGO_PROFILE_DEV_DEBUG='0';CARGO_PROFILE_TEST_DEBUG='0';CARGO_NET_OFFLINE='true';RUST_TEST_THREADS='1'};
    checks=@(); cleanup='pending'; failure=$null
}
$state = @{schema_version=2;workspace=$qa;purpose='issue236-native-usage';test_owned=$true;processes=@{};plan=@{database=@{host='127.0.0.1';port=$port;user=$dbUser;name='postgres'}}}
$ownership = Join-Path $qa 'ownership.json'
function Save { [IO.File]::WriteAllText($receiptPath,($receipt | ConvertTo-Json -Depth 30)+"`n") }
function Check([string]$Name,[string]$Log,[int]$Code,[string]$Command,[int]$Expected=0,[string]$Filter='') {
    $counts = [ordered]@{passed=0;failed=0;ignored=0;measured=0;filtered=0;summaries=0}
    $logText = [string](Get-Content -LiteralPath $Log -Raw)
    foreach ($m in [regex]::Matches($logText,'test result: \w+\. (\d+) passed; (\d+) failed; (\d+) ignored; (\d+) measured; (\d+) filtered out;')) {
        $counts.passed += [int]$m.Groups[1].Value
        $counts.failed += [int]$m.Groups[2].Value
        $counts.ignored += [int]$m.Groups[3].Value
        $counts.measured += [int]$m.Groups[4].Value
        $counts.filtered += [int]$m.Groups[5].Value
        $counts.summaries++
    }
    $passedNames = @([regex]::Matches($logText,'(?m)^test (\S+) \.\.\. ok\s*$') | ForEach-Object { $_.Groups[1].Value })
    $check = [ordered]@{name=$Name;command=$Command;exit_code=$Code;log=$Log;sha256=(Get-FileHash -LiteralPath $Log).Hash.ToLowerInvariant();counts=$counts;passed_test_names=$passedNames}
    $receipt.checks += $check
    Save
    Write-Output "$Name exit=$Code passed=$($counts.passed) failed=$($counts.failed) ignored=$($counts.ignored) filtered=$($counts.filtered)"
    if ($Code -or ($Expected -gt 0 -and ($counts.passed -ne $Expected -or $counts.failed -ne 0 -or $counts.ignored -ne 0 -or $passedNames.Count -ne $Expected -or @($passedNames | Sort-Object -Unique).Count -ne $Expected))) { throw "$Name did not satisfy its native regression requirement." }
    if ($Filter -and @($passedNames | Where-Object { $_ -notmatch [regex]::Escape($Filter) }).Count) { throw 'Unexpected tests entered the focused lane.' }
}
Save
$secret = $null
try {
    $secret = [Convert]::ToHexString([Security.Cryptography.RandomNumberGenerator]::GetBytes(32)).ToLowerInvariant()
    $passwordPath = Join-Path $qa 'credentials/postgres-password.txt'
    $passfile = Join-Path $qa 'credentials/pgpass.conf'
    [IO.File]::WriteAllText($passwordPath,$secret,[Text.UTF8Encoding]::new($false))
    [IO.File]::WriteAllText($passfile,"127.0.0.1:${port}:*:${dbUser}:$secret`n",[Text.UTF8Encoding]::new($false))
    $log = Join-Path $PSScriptRoot "$prefix-initdb.log"
    & (Join-Path $pg 'initdb.exe') -D (Join-Path $qa 'database') -U $dbUser --auth=scram-sha-256 --encoding=UTF8 --locale=C --pwfile=$passwordPath *> $log
    Check 'owned-postgresql-initialization' $log $LASTEXITCODE 'initdb --auth=scram-sha-256 (private pwfile)'
    $state.processes.postgres = Start-LocalOwnedProcess -Role 'postgres' -Workspace $qa -FilePath (Join-Path $pg 'postgres.exe') -ArgumentList @('-D',(Join-Path $qa 'database'),'-h','127.0.0.1','-p',"$port") -WorkingDirectory $qa -LogDirectory (Join-Path $qa 'logs') -Environment @{}
    Save-LocalStackState -Path $ownership -State $state -Workspace $qa
    $ready = $false
    $deadline = [DateTime]::UtcNow.AddSeconds(30)
    do {
        & (Join-Path $pg 'pg_isready.exe') -h 127.0.0.1 -p $port -U $dbUser *> $null
        if ($LASTEXITCODE -eq 0) { $ready=$true; break }
        Start-Sleep -Milliseconds 200
    } while ([DateTime]::UtcNow -lt $deadline)
    if (!$ready -or !(Test-LocalOwnedProcess -Record $state.processes.postgres -Workspace $qa)) { throw 'Owned PostgreSQL readiness failed.' }
    $listeners = @(Get-NetTCPConnection -State Listen -LocalPort $port | Select-Object -ExpandProperty OwningProcess -Unique)
    if ($listeners.Count -ne 1 -or $listeners[0] -ne $state.processes.postgres.pid) { throw 'Database listener owner mismatch.' }
    $env:PGPASSFILE = Join-Path $qa 'credentials/nonexistent.pgpass'
    $env:PGPASSWORD = 'invalid-owned-fixture-probe'
    & (Join-Path $pg 'psql.exe') -X -w -h 127.0.0.1 -p $port -U $dbUser -d postgres -c 'SELECT 1' *> $null
    if ($LASTEXITCODE -eq 0) { throw 'Incorrect database password was accepted.' }
    Remove-Item -LiteralPath Env:PGPASSWORD
    $env:PGPASSFILE = $passfile
    & (Join-Path $pg 'psql.exe') -X -w -h 127.0.0.1 -p $port -U $dbUser -d postgres -c 'SELECT 1' *> $null
    if ($LASTEXITCODE) { throw 'Owned database authentication failed.' }
    $receipt.authentication = @{method='scram-sha-256';wrong_password_rejected=$true;authenticated_query=$true;delivery='Private fixture credential files for PostgreSQL clients. SQLx environment-only delivery is reduced assurance.'}
    Save
    $env:DATABASE_URL = "postgres://${dbUser}:${secret}@127.0.0.1:$port/postgres"
    Push-Location -LiteralPath $product
    try {
        $lanes = @(@{package='crony-store';filter='issue236_';expected=16;name='usage-provenance-transactions'})
        if ($Phase -eq 'green') {
            $lanes += @{package='crony-store';filter='control_accounting_tests::issue87_';expected=14;name='hard-control-transactions'}
        }
        foreach ($lane in $lanes) {
            $listLog = Join-Path $PSScriptRoot "$prefix-$($lane.package)-$($lane.filter.Replace('::','-').TrimEnd('_'))-list.log"
            & cargo test --locked --offline -p $($lane.package) $lane.filter -- --ignored --list 2>&1 | ForEach-Object { ([string]$_).Replace($secret,'[ephemeral database credential]') } | Set-Content -LiteralPath $listLog -Encoding utf8
            if ($LASTEXITCODE) { throw 'Could not discover the complete focused ignored test lane.' }
            $discovered = @([regex]::Matches((Get-Content -LiteralPath $listLog -Raw),'(?m)^(\S+): test\s*$') | ForEach-Object { $_.Groups[1].Value })
            if ($discovered.Count -eq 0 -or ($lane.expected -gt 0 -and $discovered.Count -ne $lane.expected) -or @($discovered | Where-Object { $_ -notmatch [regex]::Escape($lane.filter) }).Count) { throw 'Native test discovery differs from the reviewed expected lane.' }
            $expected = $discovered.Count
            $log = Join-Path $PSScriptRoot "$prefix-$($lane.package)-$($lane.filter.Replace('::','-').TrimEnd('_')).log"
            & cargo test --locked --offline -p $($lane.package) $lane.filter -- --ignored --test-threads=1 2>&1 | ForEach-Object { ([string]$_).Replace($secret,'[ephemeral database credential]') } | Set-Content -LiteralPath $log -Encoding utf8
            $testCode = $LASTEXITCODE
            Check $lane.name $log $testCode "cargo test --locked --offline -p $($lane.package) $($lane.filter) -- --ignored --test-threads=1" $expected $lane.filter
            $laneReceipt = $receipt.checks[-1]
            $laneReceipt.package = $lane.package
            $laneReceipt.discovery_log = $listLog
            $laneReceipt.discovery_sha256 = (Get-FileHash -LiteralPath $listLog).Hash.ToLowerInvariant()
            $laneReceipt.discovered_test_names = $discovered
            $laneReceipt.complete_discovered_lane_executed = !(Compare-Object @($discovered | Sort-Object) @($laneReceipt.passed_test_names | Sort-Object))
            if (!$laneReceipt.complete_discovered_lane_executed) { throw 'Executed test set differs from discovery.' }
            Save
        }
    } finally { Pop-Location }
    $receipt.status = 'passed'
} catch {
    $receipt.status = 'failed'
    $message = [regex]::Replace($_.Exception.Message,'\bpostgres(?:ql)?://\S+','[database URL withheld]')
    if ($secret) { $message=$message.Replace($secret,'[ephemeral credential withheld]') }
    $receipt.failure = $message
} finally {
    try {
        $afterText = & $node (Join-Path $PSScriptRoot 'issue236-source-snapshot-r1.mjs') $product (Join-Path $PSScriptRoot "$prefix-source-after.json")
        if ($LASTEXITCODE) { throw 'Cannot capture final native test source.' }
        $receipt.source_after = $afterText | ConvertFrom-Json -AsHashtable
        $receipt.source_unchanged = $receipt.source_after.source_fingerprint -ceq $before.source_fingerprint -and $receipt.source_after.head -ceq $ExpectedHead
        if (!$receipt.source_unchanged) { throw 'Native source changed during validation.' }
        if ((Get-FileHash -LiteralPath $ReferenceManifest).Hash.ToLowerInvariant() -cne $referenceHash -or
            $receipt.source_after.status -cne $reference.status) { throw 'Reviewed reference manifest or source status changed.' }
    } catch { $receipt.status='failed'; $receipt.source_failure=$_.Exception.Message }
    Remove-Item -LiteralPath Env:DATABASE_URL,Env:PGPASSFILE,Env:PGPASSWORD -ErrorAction SilentlyContinue
    $secret = $null
    if ($state.processes.postgres) {
        try {
            if (!(Test-LocalOwnedProcess -Record $state.processes.postgres -Workspace $qa)) { throw 'Ownership cannot be verified.' }
            $log = Join-Path $PSScriptRoot "$prefix-stop.log"
            & (Join-Path $pg 'pg_ctl.exe') -D (Join-Path $qa 'database') -m fast -w stop *> $log
            if ($LASTEXITCODE) { throw 'Owned PostgreSQL stop failed.' }
            if (@(Get-NetTCPConnection -State Listen | Where-Object LocalPort -eq $port).Count) { throw 'Fixture port still has a listener after owned stop.' }
            $state.stopped_at = [DateTime]::UtcNow.ToString('o')
            Save-LocalStackState -Path $ownership -State $state -Workspace $qa
            $receipt.cleanup = 'Exact owned PostgreSQL stopped and fixture port verified inactive; database, private credentials and logs preserved.'
        } catch { $receipt.status='failed'; $receipt.cleanup='Ownership or cleanup could not be verified; preserve fixture.' }
    } else { $receipt.cleanup='No database process created; fixture files retained.' }
    $receipt.finished_at_utc = [DateTime]::UtcNow.ToString('o')
    Save
}
[pscustomobject]$receipt | Select-Object issue,status,source_unchanged,cleanup,failure | ConvertTo-Json
if ($receipt.status -ne 'passed') { exit 1 }
