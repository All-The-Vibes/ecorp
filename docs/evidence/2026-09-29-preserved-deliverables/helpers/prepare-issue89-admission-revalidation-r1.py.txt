def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(name):
    return json.loads((EV / name).read_text(encoding="utf-8-sig"))


def write_new(name, text):
    with (EV / name).open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(text)


def test_receipt(name):
    value = load(name + ".json")
    assert value["source_unchanged"] and value["source_before"] == value["source_after"]
    log = EV / (name + ".log")
    assert digest(log) == value["log_sha256"]
    text = log.read_text(encoding="utf-8")
    matches = re.findall(
        r"test result: (?:ok|FAILED)\. (\d+) passed; (\d+) failed; (\d+) ignored; (\d+) measured; (\d+) filtered out;",
        text,
    )
    assert matches, "Actual Rust result summaries are required"
    counts = dict(zip(["passed", "failed", "ignored", "measured", "filtered_out"],
                      (sum(int(row[i]) for row in matches) for i in range(5))))
    return value, {"receipt": name + ".json", "receipt_sha256": digest(EV / (name + ".json")),
                   "log": str(log), "log_sha256": value["log_sha256"], "command": value["command"],
                   "exit_code": value["exit_code"], "counts": counts, "summaries": len(matches),
                   "source_unchanged": True}
