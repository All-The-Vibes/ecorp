#requires -Version 7.5
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $false
$repo = '<USERPROFILE>\.codex\worktrees\issue219-publication\ecorp'
$expected = 'b3fbd3bcf5bd0aeef032ca0ea7ed2ca5841b07ab'
$receiptPath = Join-Path $PSScriptRoot 'issue219-canonical-invocation-r7.json'
if (Test-Path -LiteralPath $receiptPath) { throw 'Preserve the existing invocation.' }
$target = [IO.Path]::GetFullPath((Join-Path $repo 'target'))
$key = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.Encoding]::UTF8.GetBytes($target.ToLowerInvariant())))
$mutex = [Threading.Mutex]::new($false,"Local\ECorpCompletionCargo$key")
$acquired = $false
$record = [ordered]@{issue=219;repository=$repo;started_at_utc=[DateTime]::UtcNow.ToString('o');status='preparing';purpose='Canonical validation of durable retirement for deterministic publication prerequisite denials, with transient database and pending-audit failures remaining retryable'}
try {
    try { $acquired = $mutex.WaitOne(0) } catch [Threading.AbandonedMutexException] { $acquired = $true }
    if (!$acquired) { throw 'An active validation owns this target.' }
    if ((& git -C $repo rev-parse HEAD).Trim() -cne $expected -or (& git -C $repo branch --show-current).Trim() -cne 'codex/issue219-publication-intent') { throw 'Unexpected source identity.' }
    $paths = @(& git -C $repo ls-files --cached --others --exclude-standard | Sort-Object -Unique)
    foreach ($test in @(@{file='issue219-denial-green-r1.json';count=41},@{file='issue219-denial-correction-r1.json';count=15},@{file='issue219-denial-http-r1.json';count=9})) {
        $native = Get-Content -LiteralPath (Join-Path $PSScriptRoot $test.file) -Raw | ConvertFrom-Json
        if ($native.status -cne 'passed' -or !$native.source_unchanged -or $native.source_head -cne $expected) { throw "Native regression is not verified: $($test.file)" }
        $checkName = if ($test.file -eq 'issue219-denial-http-r1.json') { 'human-publication-http-authorization' } else { 'human-publication-request-and-execution' }
        $result = @($native.checks | Where-Object name -eq $checkName)
        if ($result.Count -ne 1 -or $result[0].exit_code -ne 0 -or $result[0].counts.passed -ne $test.count -or $result[0].counts.failed -ne 0 -or $result[0].counts.ignored -ne 0) { throw 'Native regression counts are not complete.' }
        if ($paths.Count -ne $native.source_files.Count -or @(Compare-Object $paths @($native.source_files.path)).Count) { throw 'Native regression source inventory changed.' }
        foreach ($item in $native.source_files) {
            if ((Get-FileHash -LiteralPath (Join-Path $repo $item.path)).Hash.ToLowerInvariant() -cne $item.sha256) { throw "Native regression source changed: $($item.path)" }
        }
    }
    $buildPath = Join-Path $PSScriptRoot 'issue219-acceptance-build-r7.json'
    $build = Get-Content -LiteralPath $buildPath -Raw | ConvertFrom-Json
    $browser = Get-Content -LiteralPath (Join-Path $PSScriptRoot 'issue219-browser-r13.json') -Raw | ConvertFrom-Json
    if ($build.status -cne 'passed' -or !$build.source_unchanged -or $build.source.head -cne $expected -or
        $browser.status -cne 'passed' -or !$browser.source_and_binaries_unchanged -or
        !$browser.fixture_report.acceptance.same_watcher_recovered_after_busy) { throw 'Complete browser/native acceptance is not verified.' }
    if ($paths.Count -ne $build.source.files.Count -or @(Compare-Object $paths @($build.source.files.path)).Count) { throw 'Acceptance source inventory changed.' }
    foreach ($item in $build.source.files) {
        if ((Get-FileHash -LiteralPath (Join-Path $repo $item.path)).Hash.ToLowerInvariant() -cne $item.sha256) { throw "Acceptance source changed: $($item.path)" }
    }
    $changes = @(& git -C $repo diff --cached --name-only)
    $expectedChanges = @('crates/crony-store/src/admission.rs','crates/crony-store/src/budget_checkpoint.rs','crates/crony-store/src/checkpoint_correction.rs','crates/crony-store/src/checkpoint_correction_history.rs','crates/crony-store/src/checkpoint_correction_publication_tests.rs','crates/crony-store/src/checkpoint_publication.rs','crates/crony-store/src/checkpoint_retention.rs','crates/crony-store/src/factory_authority.rs','crates/crony-store/src/human_publication_denial_tests.rs','crates/crony-store/src/human_publication_request_tests.rs','crates/crony-store/src/lib.rs','crates/crony-store/src/publication.rs','crates/crony-store/src/publication/request.rs','crates/crony-store/src/publication/worker.rs','crates/crony-store/src/terminal_accounting.rs')
    if ($changes.Count -ne 15 -or @(Compare-Object $changes $expectedChanges).Count) { throw 'Unexpected staged source inventory.' }
    & git -C $repo diff --cached --check
    if ($LASTEXITCODE) { throw 'Staged whitespace validation failed.' }
    if (@(& git -C $repo diff --name-only).Count -or @(& git -C $repo ls-files --others --exclude-standard).Count) { throw 'Source is not completely staged.' }
    $record.head = $expected
    $record.staged_tree = (& git -C $repo write-tree).Trim()
    $record.source_file_count = $paths.Count
    $record.staged_paths = $changes
    $record.source_binding = $buildPath
    $record.source_binding_sha256 = (Get-FileHash -LiteralPath $buildPath).Hash.ToLowerInvariant()
    $record.status = 'running'
    [IO.File]::WriteAllText($receiptPath,($record | ConvertTo-Json -Depth 10)+"`n")
    & (Join-Path $PSScriptRoot 'run-canonical-r4.ps1') -RepositoryPath $repo -Issue 219 -Revision r7
    $record.canonical_exit_code = $LASTEXITCODE
    $canonical = Get-Content -LiteralPath (Join-Path $PSScriptRoot 'issue219-canonical-r7.json') -Raw | ConvertFrom-Json
    if ($record.canonical_exit_code -or $canonical.status -cne 'passed') { throw 'Canonical validation failed; preserve its actual results.' }
    if ((& git -C $repo write-tree).Trim() -cne $record.staged_tree) { throw 'Staged source changed during validation.' }
    $record.status = 'passed'
} catch {
    $record.status = 'failed'
    $record.failure = $_.Exception.Message
} finally {
    $record.finished_at_utc = [DateTime]::UtcNow.ToString('o')
    [IO.File]::WriteAllText($receiptPath,($record | ConvertTo-Json -Depth 10)+"`n")
    if ($acquired) { $mutex.ReleaseMutex() }
    $mutex.Dispose()
}
[pscustomobject]$record | Select-Object issue,status,head,staged_tree,canonical_exit_code,failure | ConvertTo-Json
if ($record.status -cne 'passed') { exit 1 }
