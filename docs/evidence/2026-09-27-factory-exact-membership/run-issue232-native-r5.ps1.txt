function Wait-Ready([scriptblock]$Probe,[string]$Description) {
    $deadline=[DateTime]::UtcNow.AddSeconds(60)
    do { try { if (& $Probe) { return } } catch { }; Start-Sleep -Milliseconds 250 } while ([DateTime]::UtcNow -lt $deadline)
    throw "$Description did not become ready."
}
