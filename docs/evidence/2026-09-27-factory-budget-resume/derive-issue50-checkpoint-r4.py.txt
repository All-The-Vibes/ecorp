def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def bind(path):
    return {'path': str(path), 'sha256': sha(path)}


def replace_once(text, old, new):
    assert text.count(old) == 1, old
    return text.replace(old, new, 1)
