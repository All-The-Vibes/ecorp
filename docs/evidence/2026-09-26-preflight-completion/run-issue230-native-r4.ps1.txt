function Save {$receipt | ConvertTo-Json -Depth 35 | Set-Content -LiteralPath $receiptPath -Encoding utf8}
function Invoke-Phase([string]$Name,[string[]]$Arguments,[int]$TimeoutSeconds=1200) {
    $stdout=Join-Path $PSScriptRoot ('issue230-native-r4-'+$Name+'.stdout.log')
    $stderr=Join-Path $PSScriptRoot ('issue230-native-r4-'+$Name+'.stderr.log')
    if((Test-Path -LiteralPath $stdout) -or (Test-Path -LiteralPath $stderr)){throw 'Preserve phase output.'}
    foreach($arg in $Arguments){if($arg.Contains('"') -or $arg.Contains("`n") -or $arg.EndsWith('\')){throw 'Unsupported phase argument.'}}
    $quoted=@($Arguments | ForEach-Object {'"'+$_+'"'})
    $phase=[ordered]@{name=$Name;started_at_utc=[DateTimeOffset]::UtcNow.ToString('o');stdout=$stdout;stderr=$stderr;status='starting'}
    $receipt.phases+=@($phase);Save
    $child=Start-Process -FilePath $pwsh -ArgumentList $quoted -WindowStyle Hidden -PassThru -RedirectStandardOutput $stdout -RedirectStandardError $stderr
    try {
        [void]$child.Handle
        $phase.pid=$child.Id;$phase.process_started_utc=$child.StartTime.ToUniversalTime().ToString('o');$phase.status='running';Save
        $deadline=[DateTimeOffset]::UtcNow.AddSeconds($TimeoutSeconds)
        while(!$child.WaitForExit(1000)){
            if([DateTimeOffset]::UtcNow -ge $deadline){
                $phase.status='timed-out';Save
                $child.Kill()
                if(!$child.WaitForExit(10000)){throw 'Owned phase termination is unverified.'}
                throw "Bounded direct-process wait exceeded for $Name; retain all logs."
            }
        }
        $child.Refresh()
        $phase.exit_code=$child.ExitCode;$phase.status='exited';$phase.finished_at_utc=[DateTimeOffset]::UtcNow.ToString('o');Save
        return [int]$child.ExitCode
    } finally {$child.Dispose()}
}
