def require(ok, message):
    if not ok:
        raise RuntimeError(message)


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def git(*args):
    return helpers.git(WT, *args).decode().strip()


def normalize(value):
    replacements = [
        (str(WT), '<candidate-checkout>'),
        (str(RUN), '<run-records>'), (str(HISTORY), '<prior-run-records>'),
        (str(QA), '<browser-fixture>'),
        (r'<native-build-fixture>', '<native-build-fixture>'),
        (r'<native-database-fixture>', '<native-database-fixture>'),
        (r'<owned-qa-root>', '<owned-qa-root>'),
        (r'<pinned-toolchain>', '<pinned-toolchain>'),
        (r'<operator-home>', '<operator-home>'),
    ]
    for original, replacement in replacements:
        for variant in (original.replace('\\', '\\\\'), original, original.replace('\\', '/')):
            expression = ''.join(r'\^*' + re.escape(character) for character in variant)
            value = re.sub(expression, lambda match: replacement, value, flags=re.IGNORECASE)
    value = re.sub(r'(?<!\w)<local-user>(?!\w)', '<local-user>', value, flags=re.IGNORECASE)
    # These are copied historical assertions/driver examples, never executable
    # product fixtures. Their unmodified originals and hashes remain retained.
    value = value.replace('<synthetic-user>', '<synthetic-user>').replace('<synthetic-user>', '<synthetic-user>')
    lines = [line.rstrip(' \t') for line in value.splitlines()]
    while lines and not lines[-1]:
        lines.pop()
    return ('\n'.join(lines) + '\n') if lines else ''


def write(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    if not isinstance(value, bytes):
        if not isinstance(value, str):
            value = json.dumps(value, indent=2, ensure_ascii=False) + '\n'
        value = normalize(value).encode('utf-8')
    with path.open('xb') as stream:
        stream.write(value)


def bound_receipt(name):
    value = read(RUN / name)
    require(value['status'] == 'passed' and value['source_unchanged'] is True,
            'Passing unchanged-source validation required: ' + name)
    for side in ('source_before', 'source_after'):
        source = value[side]
        require(source['source_fingerprint'] == FINGERPRINT
                and sha(source['manifest']) == source['manifest_sha256'],
                'Validation source binding changed: ' + name)
    return value
